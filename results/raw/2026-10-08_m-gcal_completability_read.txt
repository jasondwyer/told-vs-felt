R = a["records"]
assert len(R) == 128
clean_task = {r["i"]: r["task"] for r in R if r["pool"] == "CORR_clean"}
render = {r["i"]: r["task"] for r in R if r["pool"] == "CORR_corrupted"}
def words(s): return re.findall(r"[A-Za-z]+", s)
# words of the render that are not words of the clean task (case-insensitive): the corrupted words

      "sum", sum(r["n_tok"] for r in R))

      [(r["pool"], r["seed_index"], r["text"]) for r in close])

          f"unchanged in the render: {bool(re.search(rf'\b{g_}\b', render[i_]))}")

    need_abs = next(h for h in range(65) if Fraction(h, 64) >= Fraction(7, 10))
