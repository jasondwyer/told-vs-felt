@contextlib.contextmanager
def olmo_stub_startup():

    kit = m_testing.olmo_kit()
    setup = runner_olmo.OLMoSetup(audit=kit["audit"], ban_ids=kit["ban_ids"], ban_strings=kit["ban_strings"],

def factory_for(gate, rule=None):
    def factory():
        a = m_testing.OLMoScriptedAdapter("p-stage-selftest")
        cells = {}
        for t, n_t in stage.templates_counts(unit, gate):
            for j, cell in enumerate(m_schedule.gate_schedule(unit, gate, t, m_schedule.roster_for(unit), n_t)):
                cells[(t, j)] = cell
        a.load(m_stages._script_policy(unit, gate, cells, rule), None)
        return a
    return factory


def mask(x):
    return json.loads(re.sub(r'"sha256": "[0-9a-f]{64}"', '"sha256": "*"', json.dumps(x, sort_keys=True)))


def run_gate_case(r, gate, rule=None):
    m_scaffold._of_record.cache_clear()
    buf = io.StringIO()
    err = None
    res = None
    with olmo_stub_startup(), contextlib.redirect_stdout(buf):
        try:
            res = stage.run_gate(unit, gate, factory_for(gate, rule), stub=True, out_root=r)
        except (SystemExit, Exception) as e:
            res, err = None, e
    pp = stage.out_paths(unit, r, gate=gate)
    events = ([json.loads(ln) for ln in pp["p_calls"].read_text(encoding="utf-8").splitlines() if ln.strip()]
              if pp["p_calls"].exists() else None)
    ck = json.loads(pp["m_ckpt"].read_text(encoding="utf-8")) if pp["m_ckpt"].exists() else {}
    sessions = [{"template": s["template"], "s": s["s"], "contrast": s["contrast"], "verified": s.get("verified"),
                 "exclusion": s.get("exclusion"),
                 "rounds": [{k: rd.get(k) for k in ("phase", "ri", "ci", "seed", "text", "cap_hit")}
                            for rd in s.get("rounds", ())]} for s in ck.get("sessions", [])]
    rep = json.loads(pp["p_report"].read_text(encoding="utf-8")) if pp["p_report"].exists() else None
    stopped = pp["p_stopped"].read_text(encoding="utf-8").rstrip() if pp["p_stopped"].exists() else None
    lines = [re.sub(r"[0-9a-f]{64}", "*", ln) for ln in stage.summary_lines(res)] if res else None
    stdout = [re.sub(r"[0-9a-f]{64}", "*", ln) for ln in buf.getvalue().splitlines()]
    return mask({"status": res and res["status"], "error": err and f"{type(err).__name__}: {err}",
                 "events": events, "sessions": sessions, "identity": ck.get("identity"), "stage_end": ck.get("stage_end"),
                 "report": rep, "stopped": stopped, "summary_lines": lines, "stdout_lines": len(stdout)})


def run_case(tag, qual_rule=None, time_rule=None, qual=True):
    td = Path(tempfile.mkdtemp(prefix=f"p_gates_regress_{tag}_"))
    try:
        r = td / "p"
        p_schedule.write_schedule_files(r / "schedules")
        out = {"time": run_gate_case(r, "time", time_rule)}
        if qual:
            out["qual"] = run_gate_case(r, "qual", qual_rule)
        out["names"] = sorted(x.relative_to(r).as_posix() for x in r.rglob("*")
                              if x.is_file() and "schedules" not in x.parts)
        return out
    finally:
        shutil.rmtree(td, ignore_errors=True)
