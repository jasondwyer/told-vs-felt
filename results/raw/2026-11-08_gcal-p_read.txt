from fractions import Fraction
from pathlib import Path

ROOT = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(ROOT))
from valence_core.m import corr as m_corr  # noqa: E402
from valence_core.m import gcal as m_gcal  # noqa: E402
from valence_core.p import gcal as p_gcal  # noqa: E402
from valence_core.p import report as p_report  # noqa: E402

CKS = ("sft", "dpo", "instruct")
POOLS = ("CORR_clean", "CORR_corrupted")
CLEAN_FIELDS = ("seed", "task", "text", "n_tok", "cap_hit", "stop_reason", "hit")


def load(rel: str):
    b = (ROOT / rel).read_bytes()
    return json.loads(b), hashlib.sha256(b).hexdigest()


def key(r) -> str:
    return f"{r['pool']} i={r['i']} s={r['seed_index']}"


def frac(x: Fraction) -> str:
    return f"{x.numerator}/{x.denominator}"


pool = m_corr.load_pool()
assert len(pool) == 16


def frame_class(i: int) -> str:
    t = pool[i]["task"]
    if t.startswith("Compute"):
        return "Compute"
    if t.startswith("Write the opposite of"):
        return "Write"
    if t.startswith("State") or t.startswith("Name"):
        return "State/Name"
    return "other"


CLASSES = ("Compute", "Write", "State/Name")
P, M = {}, {}
for ck in CKS:
    P[ck] = {"ckpt": load(f"data/p/ckpt/G-CAL_olmo-{ck}.json"), "art": load(f"data/p/gates/G-CAL_completability_olmo-{ck}.json")}
    M[ck] = {"ckpt": load(f"data/m/ckpt/G-CAL_olmo-{ck}.json"), "art": load(f"data/m/gates/G-CAL_completability_olmo-{ck}.json")}

print("## A. Files read (sha256)")
for ck in CKS:
    for side, d in (("p", P), ("m", M)):
        print(f"   {side} {ck:8s} checkpoint {d[ck]['ckpt'][1]}  artifact {d[ck]['art'][1]}")
print()

print("## B. The artifact's figures against Code's recomputation from the checkpoint's records")
for ck in CKS:
    ckd, art = P[ck]["ckpt"][0], P[ck]["art"][0]
    recs = ckd["records"]
    arecs = art.get("records")
    print(f"-- OLMo-{ck}: verdict {art['verdict']}; {len(recs)} checkpoint records; artifact records "
          f"{'identical to the checkpoint records' if arecs == recs else 'DIFFER from the checkpoint records'}")
    regrade = [key(r) for r in recs if m_gcal.grade(r["text"], pool[r["i"]]["gold"]) != r["hit"]]
    print(f"   grader re-applied (m_gcal.grade on the stored text): {len(recs) - len(regrade)}/{len(recs)} agree with the "
          f"record's hit field {regrade if regrade else ''}")
    keys = [(r["pool"], r["i"], r["seed_index"]) for r in recs]
    want = [(p, i, s) for p in POOLS for i in range(16) for s in range(4)]
    print(f"   keys: {len(set(keys))} distinct; order and set as m:176's (pool, i, seed 0..3): {keys == want}")
    acc = {}
    for p in POOLS:
        rs = [r for r in recs if r["pool"] == p]
        h = sum(1 for r in rs if r["hit"])
        acc[p] = Fraction(h, len(rs))
        a = art["accuracy"][p]
        print(f"   {p:15s} {h}/{len(rs)} = {frac(acc[p])}  (artifact {a['hits']}/{a['n']} = {a['fraction']} "
              f"{a['decimal_4dp']}: {'match' if (a['hits'], a['n'], a['fraction']) == (h, len(rs), frac(acc[p])) else 'MISMATCH'})")
    ac, ax = acc["CORR_clean"], acc["CORR_corrupted"]
    absp, relbar = ax >= Fraction(7, 10), Fraction(3, 4) * ac
    relp = ax >= relbar
    v = "PASS" if (absp and relp) else "FAIL"
    c4 = art["condition_4"]
    print(f"   condition (4): corrupted {frac(ax)} >= 7/10: {'yes' if absp else 'no'}; >= 3/4 x clean = {frac(relbar)}: "
          f"{'yes' if relp else 'no'} -> {v}  (artifact {c4['verdict']}, rel_bar {c4['rel_bar']}: "
          f"{'match' if (c4['verdict'], c4['rel_bar']) == (v, frac(relbar)) else 'MISMATCH'})")
    hits_needed = max(math.ceil(Fraction(7, 10) * 64), math.ceil(Fraction(3, 4) * ac * 64))
    print(f"   hits needed on the corrupted pool at this clean count: {hits_needed}/64; observed "
          f"{sum(1 for r in recs if r['pool'] == 'CORR_corrupted' and r['hit'])}/64")
    cov = ac - ax
    print(f"   covariate (clean minus corrupted): {frac(cov)} = {float(cov):+.4f}  (artifact {art['covariate']['fraction']} "
          f"{art['covariate']['decimal_4dp']}: {'match' if art['covariate']['fraction'] == frac(cov) else 'MISMATCH'})")
    # clean-pool identity, recomputed against G-CAL-OLMo's artifact records and checkpoint records
    for label, ref in (("G-CAL-OLMo artifact", M[ck]["art"][0]), ("G-CAL-OLMo checkpoint", M[ck]["ckpt"][0])):
        by = {(r["i"], r["seed_index"]): r for r in ref["records"] if r["pool"] == "CORR_clean"}
        diff = [key(r) for r in recs if r["pool"] == "CORR_clean"
                and any(r[f] != by[(r["i"], r["seed_index"])][f] for f in CLEAN_FIELDS)]
        print(f"   clean-pool identity vs {label}: {64 - len(diff)}/64 identical on {CLEAN_FIELDS} {diff if diff else ''}")
    ci = art["p"]["clean_pool_identity"]
    print(f"   artifact's clean_pool_identity: {ci['identical']}/{ci['n']} differing {ci['differing']}; generation path "
          f"fields equal: {all(x['equal'] for x in ci['generation_path'].values())} "
          f"{[f for f, x in ci['generation_path'].items() if not x['equal']]}")
    # corrupted items whose p render equals m's: the same (prompt, seed) pairs as G-CAL-OLMo's
    mrec = {(r["pool"], r["i"], r["seed_index"]): r for r in M[ck]["ckpt"][0]["records"]}
    same_items = sorted({r["i"] for r in recs if r["pool"] == "CORR_corrupted"
                         and r["task"] == mrec[("CORR_corrupted", r["i"], r["seed_index"])]["task"]})
    sd = [key(r) for r in recs if r["pool"] == "CORR_corrupted" and r["i"] in same_items
          and any(r[f] != mrec[("CORR_corrupted", r["i"], r["seed_index"])][f] for f in CLEAN_FIELDS)]
    print(f"   corrupted items whose p render equals m's render: {same_items}; their {4 * len(same_items)} generations "
          f"identical to G-CAL-OLMo's: {4 * len(same_items) - len(sd)}/{4 * len(same_items)} {sd if sd else ''}")
    sg = art["p"]["strict_grade"]
    mine = {p: sum(1 for r in recs if r["pool"] == p and p_gcal.strict_hit(r["text"], pool[r["i"]]["gold"])) for p in POOLS}
    print(f"   strict grade (decides nothing): clean {mine['CORR_clean']}/64, corrupted {mine['CORR_corrupted']}/64 "
          f"(artifact {sg['accuracy']['CORR_clean']['hits']}, {sg['accuracy']['CORR_corrupted']['hits']}); graded hit, "
          f"strict miss: {sg['graded_hit_strict_miss']}")
    inv = art["p"]["stored_text_invariant"]
    nl = [key(r) for r in recs if p_report.line_break_in_stored_text(r["text"])]
    print(f"   stored-text invariant: texts with a line break {len(nl)} {nl if nl else ''}; proxy cuts "
          f"{inv['n_cuts']} (checkpoint cuts list {len(ckd.get('cuts', []))})")
    print(f"   hair trigger block: {json.dumps(art['p']['hair_trigger'], ensure_ascii=False)[:300]}")
    cn = art["p"]["counters"]
    for c in p_report.COUNTERS:
        print(f"   counter {c}: {cn[c]['n']} {cn[c]['keys']}")
    extra = {k: v for k, v in cn.items() if k not in p_report.COUNTERS}
    if extra:
        print(f"   other counter fields: {extra}")
    print(f"   identity: proposal_sha256 {art['identity'].get('proposal_sha256', '')[:16]}..., proposal_p_sha256 "
          f"{art.get('proposal_p_sha256', '')[:16]}..., unit {art['unit']}, stage {art['stage']}")
    print(f"   disclosure: {art['p']['disclosure']}")
print()

print("## C. Corrupted-pool hits per item: p (frame-word dose) against G-CAL-OLMo (m's shared dose), by §3.3's classes")
hdr = "   item class       " + "".join(f"{ck:>14s}" for ck in CKS) + "   p render | m render"
print(hdr)
cls_tot = {ck: {c: [0, 0, 0] for c in CLASSES} for ck in CKS}  # p hits, m hits, n
for i in range(16):
    row = []
    pt = mt = None
    for ck in CKS:
        pr = [r for r in P[ck]["ckpt"][0]["records"] if r["pool"] == "CORR_corrupted" and r["i"] == i]
        mr = [r for r in M[ck]["ckpt"][0]["records"] if r["pool"] == "CORR_corrupted" and r["i"] == i]
        ph, mh = sum(r["hit"] for r in pr), sum(r["hit"] for r in mr)
        pt, mt = pr[0]["task"], mr[0]["task"]
        row.append(f"p {ph}/4 m {mh}/4")
        c = frame_class(i)
        cls_tot[ck][c][0] += ph
        cls_tot[ck][c][1] += mh
        cls_tot[ck][c][2] += 4
    print(f"   {i:4d} {frame_class(i):10s} " + "".join(f"{x:>14s}" for x in row) + f"   {pt!r} | {mt!r}")
print("   class totals (p hits | m hits, of n):")
for c in CLASSES:
    print(f"     {c:10s} " + "; ".join(f"{ck} p {cls_tot[ck][c][0]} | m {cls_tot[ck][c][1]} of {cls_tot[ck][c][2]}" for ck in CKS))
print("   clean-pool hits per class (p = m by the identity):")
for c in CLASSES:
    print(f"     {c:10s} " + "; ".join(
        f"{ck} {sum(r['hit'] for r in P[ck]['ckpt'][0]['records'] if r['pool'] == 'CORR_clean' and frame_class(r['i']) == c)}"
        for ck in CKS))
print()

print("## D. Counts beside the counters")
for ck in CKS:
    recs = P[ck]["ckpt"][0]["records"]
    caps = {p: sum(1 for r in recs if r["pool"] == p and r["cap_hit"]) for p in POOLS}
    mark = [key(r) for r in recs if ("===" in r["text"] or "[PROLOGUE" in r["text"] or "<functions>" in r["text"]
                                     or r["text"].lstrip().startswith("###") or r["text"].lstrip().startswith("---"))]
    stops = Counter(r["stop_reason"] for r in recs)
    print(f"   {ck}: cap hits clean {caps['CORR_clean']} corrupted {caps['CORR_corrupted']}; stop reasons {dict(stops)}")
    print(f"      texts with a scaffold-like marker (===, [PROLOGUE, <functions>, leading ### or ---) -- G-CAL-OLMo's read's "
          f"class: {len(mark)} {mark}")
    ntok = [r["n_tok"] for r in recs]
    print(f"      n_tok: min {min(ntok)} max {max(ntok)} total {sum(ntok)}")
for a, b in (("sft", "dpo"), ("dpo", "instruct"), ("sft", "instruct")):
    ra = {(r["pool"], r["i"], r["seed_index"]): r["text"] for r in P[a]["ckpt"][0]["records"]}
    rb = {(r["pool"], r["i"], r["seed_index"]): r["text"] for r in P[b]["ckpt"][0]["records"]}
    same = sum(1 for k in ra if ra[k] == rb[k])
    samec = sum(1 for k in ra if k[0] == "CORR_corrupted" and ra[k] == rb[k])
    print(f"   identical text for the same key, {a} vs {b}: {same}/128 (corrupted pool {samec}/64)")
print()

print("## E. The proxy's cuts (stored text, and the adapter's returned text: a diagnostic field, never stored text)")
anycut = False
for ck in CKS:
    ckd = P[ck]["ckpt"][0]
    for c in ckd.get("cuts", []):
        anycut = True
        r = ckd["records"][c["record"]]
        print(f"   {ck} {key(r)}: stored {c.get('stored_text')!r} (n_tok {c.get('n_tok_stored')}); returned "
              f"{c.get('returned_text')!r} (n_tok {c.get('n_tok_returned')}, stop {c.get('stop_reason_returned')}, "
              f"cap {c.get('cap_hit_returned')})")
if not anycut:
    print("   none")
print()

print("## F. All 384 generations (per checkpoint, pool, item: the task as served, the gold list, then each seed's grade "
      "of record, the strict grade, token count, stop reason and stored text; a cut record adds the returned text)")
for ck in CKS:
    ckd = P[ck]["ckpt"][0]
    cuts = {c["record"]: c for c in ckd.get("cuts", [])}
    print(f"######## OLMo-{ck}")
    for p in POOLS:
        for i in range(16):
            rs = [(j, r) for j, r in enumerate(ckd["records"]) if r["pool"] == p and r["i"] == i]
            print(f"-- {p} item {i} | task: {rs[0][1]['task']!r} | gold: {pool[i]['gold']}")
            for j, r in rs:
                st = "S" if p_gcal.strict_hit(r["text"], pool[i]["gold"]) else "-"
                cap = " CAP" if r["cap_hit"] else ""
                print(f"   s{r['seed_index']} {'HIT ' if r['hit'] else 'miss'} strict:{st} n={r['n_tok']:2d} "
                      f"{r['stop_reason']}{cap} :: {r['text']!r}")
                if j in cuts:
                    print(f"      CUT -- returned (diagnostic): {cuts[j].get('returned_text')!r}")

print()
print("## G. R1 A6's four tallies by m's own function (valence_core/m/gcal.py olmo_tallies, scaffold.load_term_rx), on p's "
      "records with p's renders -- POST-HOC, descriptive, for comparability with G-CAL-OLMo's entry only; p's spec does not "
      "declare them (its counters are section B's)")
from valence_core.m import scaffold as m_sc  # noqa: E402
from valence_core.p import corrupt as p_corrupt  # noqa: E402
items = []
for i, it in enumerate(pool):
    rend, meta = p_corrupt.corrupt_p(it["task"], i, m_corr.ROSTERS_OF_RECORD)
    assert rend == p_corrupt.RENDERS_OF_RECORD[i], (i, rend)
    items.append({"i": i, "render": rend, "changed_positions": list(meta["changed_positions"])})
print("   p's corrupted words per item (m_gcal.corrupted_words on p's renders): "
      + "; ".join(f"{d['i']}: {m_gcal.corrupted_words(d)}" for d in items))
term_rx = m_sc.load_term_rx()
for ck in CKS:
    t = m_gcal.olmo_tallies(P[ck]["ckpt"][0]["records"], items, term_rx)
    print(f"   {ck}: (iii) {t['iii_termination']['n']} {t['iii_termination']['keys']} | (iv) {t['iv_mirroring']['n']}, of them "
          f"graded hits {t['iv_mirroring']['hits']} {t['iv_mirroring']['keys']} | (i) candidates {t['i_candidates']['n']} | "
          f"(ii) candidates {t['ii_candidates']['n']} {t['ii_candidates']['keys']}")
print()
print("## H. Texts mentioning configuration, alignment or evaluation (case-insensitive 'config', 'align', 'evaluat'), every "
      "record")
import re as _re  # noqa: E402
for ck in CKS:
    for r in P[ck]["ckpt"][0]["records"]:
        if _re.search(r"config|align|evaluat", r["text"], _re.IGNORECASE):
            print(f"   {ck} {key(r)}: {r['text']!r}")
