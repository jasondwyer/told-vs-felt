"""D3 (the seat's ruling, 2026-10-10): G-SHAPE-F0's sources map compared with the tree (PIN-S10).

Read-only. Compares data/m/gates/G-SHAPE-F0.json's provenance `sources` and `tooling` maps with
(1) gshape.sources_map() (working-tree bytes) and (2) gshape.committed_sources() (the committed blobs at
HEAD, and whether each is identical in the working tree); lists the m package's files whose bytes differ
between the G-SHAPE-F0 build (d092078) and HEAD; and prints the package_sha256 of each. No model or
tokenizer.
"""
import hashlib
import json
import subprocess
import sys
from pathlib import Path

REPO = Path("/Users/jasondwyer/Documents/code/jspace-lab")
sys.path.insert(0, str(REPO))
from valence_core.m import gshape, resume  # noqa: E402

ok_all = []


def chk(name, ok, got=""):
    ok_all.append(bool(ok))
    print(f"  [{'PASS' if ok else 'FAIL'}] {name}   {got}")


def git(*a):
    return subprocess.run(["git", "--no-optional-locks", "-C", str(REPO), *a], check=True,
                          capture_output=True, text=True).stdout


art_p = REPO / "data/m/gates/G-SHAPE-F0.json"
art_b = art_p.read_bytes()
prov = json.loads(art_b)["provenance"]
print(f"HEAD {git('rev-parse', 'HEAD').strip()}")
print(f"G-SHAPE-F0 artifact sha256 {hashlib.sha256(art_b).hexdigest()}")
print(f"G-SHAPE-F0 artifact package_sha256 {prov['package_sha256']}")
print(f"package_sha256 now (resume.package_sha256, working tree) {resume.package_sha256()}")

wt = gshape.sources_map()
probs, cm = gshape.committed_sources()
for part in ("sources", "tooling"):
    a, w, c = prov[part], wt[part], cm[part]
    diff_w = sorted(k for k in set(a) | set(w) if a.get(k) != w.get(k))
    diff_c = sorted(k for k in set(a) | set(c) if a.get(k) != c.get(k))
    chk(f"{part}: {len(a)} files in the artifact's map; working tree identical", not diff_w, diff_w)
    chk(f"{part}: committed blobs at HEAD identical to the artifact's map", not diff_c, diff_c)
chk("every map file committed at HEAD with the working tree's bytes (gshape.committed_sources)", not probs, probs[:3])
chk("sources digest equals the artifact's", gshape.sources_digest(wt) == gshape.sources_digest(
    {"sources": prov["sources"], "tooling": prov["tooling"]}), gshape.sources_digest(wt))
in_maps = set(prov["sources"]) | set(prov["tooling"])
chk("valence_core/m/stages.py is in neither map (D3: the D2 edit does not re-run G-SHAPE-F0)",
    "valence_core/m/stages.py" not in in_maps)
chk("scripts/m_analyze.py is in neither map", "scripts/m_analyze.py" not in in_maps)
changed = [ln for ln in git("diff", "--name-status", "d092078", "HEAD", "--", "valence_core/m").splitlines() if ln]
print(f"valence_core/m changes d092078..HEAD: {changed}")
chk("valence_core/m diff d092078..HEAD is stages.py alone (the lock states it, D1/D3)",
    changed == ["M\tvalence_core/m/stages.py"], changed)
chk("no SCORING_STACK_FILES entry in that diff (critic's LOCK hazard condition, D1)",
    not any(ln.split("\t", 1)[1] in gshape.SCORING_STACK_FILES for ln in changed))
n = len(ok_all)
print(f"  {sum(ok_all)}/{n} D3 checks hold")
sys.exit(0 if all(ok_all) else 1)
