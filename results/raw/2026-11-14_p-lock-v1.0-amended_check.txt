    cp, data = m_stages._committed(rel)

#!/usr/bin/env python3
"""Is each amended draft exactly 012ffbe's draft with (A1)-(A3) applied? (scratch; model-free; read-only)

Independent of the generator: the amendments' texts are cut from the seat's raw by the arrow and the backticks, not by
the generator's regex; each is applied to the draft committed at 012ffbe (git show) as one exact edit, its anchor
asserted to occur once; the result is compared with the generator's draft byte for byte. Then each draft's unified
diff against 012ffbe's (difflib, file labels only: no times) and its sha256 are printed, for the entry.

Usage: python -I -B diff_check.py <repo> <drafts_dir>"""
from __future__ import annotations

import difflib
import hashlib
import subprocess
import sys
from pathlib import Path

ROOT = Path(sys.argv[1]).resolve()
DRAFTS = Path(sys.argv[2]).resolve()
SEAT_A = ("results/raw/2026-11-14_gf3-and-lock-p-v1_design-to-code_answers.md",
          "6b963adf126540dab95780d486ab0a6652a1308828548a5cb2763f998040db3a")
BASE = "012ffbe"
UNITS = ("sft", "dpo", "instruct")

raw = (ROOT / SEAT_A[0]).read_bytes()
assert hashlib.sha256(raw).hexdigest() == SEAT_A[1], "the seat's raw is not at its sha256"
lines = raw.decode("utf-8").split("\n")


def bullet(tag: str) -> str:
    hit = [x for x in lines if x.startswith(f"- **({tag}) ")]
    assert len(hit) == 1, tag
    return hit[0]


def unquote(s: str) -> str:
    s = s.strip()
    assert s.startswith("`") and s.endswith("`") and s.count("`") == 2, s[:40]
    return s[1:-1]


a1 = bullet("A1").split("** ", 1)[1]
a1_old, a1_new = (unquote(x) for x in a1.split(" → "))
a2 = bullet("A2")
a2_text = unquote(a2[a2.index(", a bullet: ") + len(", a bullet: "):])
a3 = bullet("A3").split("** ", 1)[1]
a3_old, a3_new = (unquote(x) for x in a3.split(" → "))
assert a3_old.startswith("… ") and a3_new.startswith("… ")
a3_old, a3_new = a3_old[2:], a3_new[2:]

bad = 0
for ck in UNITS:
    rel_old = f"results/raw/2026-11-14_p-lock-v1.0-draft_OLMo-{ck}.md"
    old = subprocess.run(["git", "-C", str(ROOT), "show", f"{BASE}:{rel_old}"], check=True,
                         capture_output=True).stdout.decode("utf-8")
    new_b = (DRAFTS / f"LOCK-p-OLMo-{ck}.md").read_bytes()
    new = new_b.decode("utf-8")
    exp = old
    assert exp.count(a1_old) == 1, (ck, "A1")
    exp = exp.replace(a1_old, a1_new)
    q = [x for x in exp.split("\n") if x.startswith("  > 3. **LOCK-p v1.0 ×3** by D_p")]
    assert len(q) == 1, (ck, "A2 anchor")
    anchor = q[0] + "\n\n"
    assert exp.count(anchor) == 1, (ck, "A2 anchor once")
    exp = exp.replace(anchor, anchor + "- " + a2_text + "\n")
    assert exp.count(a3_old) == 1, (ck, "A3")
    exp = exp.replace(a3_old, a3_new)
    same = exp.encode("utf-8") == new_b
    bad += not same
    old_sha = hashlib.sha256(old.encode("utf-8")).hexdigest()
    new_sha = hashlib.sha256(new_b).hexdigest()
    print(f"OLMo-{ck}: 012ffbe's draft sha256 {old_sha}; amended draft sha256 {new_sha}; "
          f"equal to 012ffbe's draft with (A1)-(A3) applied: {same}")
    d = list(difflib.unified_diff(old.splitlines(keepends=True), new.splitlines(keepends=True),
                                  fromfile=f"{BASE}:{rel_old}", tofile=f"LOCK-p-OLMo-{ck}.md"))
    hunks = [x for x in d if x.startswith("@@")]
    plus = [x for x in d if x.startswith("+") and not x.startswith("+++")]
    minus = [x for x in d if x.startswith("-") and not x.startswith("---")]
    print(f"OLMo-{ck}: {len(hunks)} hunks, {len(minus)} lines removed, {len(plus)} lines added")
    sys.stdout.write("".join(d))
    print()
print("ALL EXACT" if bad == 0 else f"{bad} draft(s) not exact")
sys.exit(1 if bad else 0)

#!/usr/bin/env python3
"""The precheck at ``campaign`` over each LOCK-p v1.0 draft, as if the draft were the committed lock (scratch;
model-free; read-only). m_stages._committed is stood in for the lock's path alone -- it returns the draft's bytes --
and the pod id is a stand-in, since the precheck refuses without one (the box's own runs carry the real one). Nothing
is loaded, generated or written; every other check reads the committed tree as the box's precheck will.

Usage: python -I -B dry_precheck.py <repo> <drafts_dir>"""
from __future__ import annotations

import hashlib
import os
import sys
from pathlib import Path
from unittest import mock

ROOT = Path(sys.argv[1]).resolve()
DRAFTS = Path(sys.argv[2]).resolve()
sys.path.insert(0, str(ROOT))
from valence_core.m import resume as m_resume  # noqa: E402
from valence_core.m import stages as m_stages  # noqa: E402
from valence_core.p import stage  # noqa: E402

real = m_stages._committed
bad = 0
for u in stage.UNITS:
    rel = stage.lock_rel(u)
    draft = (DRAFTS / Path(rel).name).read_bytes()

    def committed(r, *a, _rel=rel, _d=draft, **kw):
        return ([], _d) if r == _rel else real(r, *a, **kw)
    with mock.patch.object(m_stages, "_committed", committed), \
            mock.patch.dict(os.environ, {m_resume.POD_ENV: "dry-precheck-stand-in"}):
        try:
            pre = stage.precheck(u, stub=False, gate="campaign")
            ok = pre["lock_bytes"] == draft and pre["backend_of_record"] == "641ee466f4553b6a"
            gpwr16 = pre["lock_required"].get("G-PWR (p)'s record", "")[:16]
            print(f"{u}: CLEAR at campaign over the draft (sha256 {hashlib.sha256(draft).hexdigest()}); lock_sha256 "
                  f"{pre['identity_extra'].get('lock_sha256')}; backend {pre['backend_of_record']}; "
                  f"{len(pre['lock_required'])} values required; G-PWR (p)'s record {gpwr16}...; {ok}")
            bad += not ok
        except SystemExit as e:
            print(f"{u}: REFUSED: {e}")
            bad += 1
print("ALL CLEAR" if bad == 0 else f"{bad} unit(s) not clear")
sys.exit(1 if bad else 0)
