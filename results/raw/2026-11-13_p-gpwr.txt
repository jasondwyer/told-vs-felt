    if k > n:
        return Fraction(0)
    return sum(comb(n, x) * p ** x * (1 - p) ** (n - x) for x in range(k, n + 1))


def crit(n, p0, alpha):
    """The smallest k with P(X >= k | n, p0) <= alpha, or None."""
    for k in range(0, n + 1):
        if sf(k, n, p0) <= alpha:
            return k
    return None


def seq_false_pause(ns, p0, alpha):
    """P(some look j has cumulative count >= crit(n_j)), increments Bin(n_j - n_{j-1}, p0): a DP over the surviving
    counts, written here from the definition."""
    dist, prev = {0: Fraction(1)}, 0
    for n in ns:
        d = n - prev
        c = crit(n, p0, alpha)
        nd = {}
        for s, w in dist.items():
            for x in range(0, d + 1):
                t = s + x
                if c is not None and t >= c:
                    continue
                nd[t] = nd.get(t, 0) + w * comb(d, x) * p0 ** x * (1 - p0) ** (d - x)
        dist, prev = nd, n
    return 1 - sum(dist.values())


def n90(floor, r):
    n = floor
    while sf(floor, n, r) < Fraction(9, 10):
        n += 1
    return n


def launch(floor, r):
    up = ceil(Fraction(115, 100) * floor)
    nn = n90(floor, r)
    return {"floor": floor, "ceil_1.15_floor": up, "N90": nn, "L": -(-max(up, nn) // UNIT) * UNIT}


def looks_of(stop):
    """The campaign rule as written: the next launch is the arm with the fewest launches still short of its stop,
    ties to A0; a look at every 40th launch and at the end."""
    cnt, tot, out = {"A0": 0, "A0s": 0}, 0, []
    while any(cnt[a] < stop[a] for a in ("A0", "A0s")):
        live = [a for a in ("A0", "A0s") if cnt[a] < stop[a]]
        a = min(live, key=lambda x: (cnt[x], 0 if x == "A0" else 1))
        cnt[a] += 1
        tot += 1
        if tot % LOOK == 0:
            out.append((tot, cnt["A0"], cnt["A0s"]))
    if not out or out[-1][0] != tot:
        out.append((tot, cnt["A0"], cnt["A0s"]))
    return out


def dec(x, places):
    scale = 10 ** places
    v = (x * scale * 2 + 1) // 2
    whole, frac = divmod(int(v), scale)
    return f"{whole}.{frac:0{places}d}"
