def amend(r, new_cells, test, text, label="the A4 sweep"):
    r["null_cells"] = new_cells
    r["rationale_by"] = f"{r['rationale_by']}, as amended by Code ({label}, {test})"
    r["rationale"] = text

def main():
    src, out = Path(sys.argv[1]), Path(sys.argv[2])
    base = json.loads(src.read_text(encoding="utf-8"))
    units = {r["unit"] for r in base}
    groups = (SEAT, SWEEP, REGROUND, APPEND)
    for d in groups:
        assert set(d) <= units, set(d) - units
    assert all(not (set(a) & set(b)) for i, a in enumerate(groups) for b in groups[i + 1:])
    rows = copy.deepcopy(base); log = []
    for r in rows:
        u, old = r["unit"], list(r["null_cells"])
        if u in SEAT:
            amend(r, *SEAT[u])
        elif u in SWEEP:
            amend(r, *SWEEP[u])
        elif u in REGROUND:
            test, text = REGROUND[u]; amend(r, old, test, text)
        elif u in APPEND:
            test, text = APPEND[u]; amend(r, old, test, r["rationale"] + " " + text)
        else:
            continue
        log.append(f"{u}: {','.join(old)} -> {','.join(r['null_cells'])} ({r['rationale_by'].rsplit('(', 1)[-1]}"
                   if r["null_cells"] != old else f"{u}: cells unchanged; rationale ({r['rationale_by'].rsplit('(', 1)[-1]}")
    p = out / "rows_of_record.v3.json"
    p.write_text(json.dumps(rows, ensure_ascii=False, indent=1) + "\n", encoding="utf-8")
    print(f"== {p.name}: {len(rows)} rows sha256 {hashlib.sha256(p.read_bytes()).hexdigest()}")
    print("\n".join("   " + l for l in log))

main()
