def pid1_pod_id() -> str | None:
    try:
        raw = open("/proc/1/environ", "rb").read().split(b"\0")
    except OSError:
        return None
    for kv in raw:
        if kv.startswith(b"RUNPOD_POD_ID="):
            return kv.split(b"=", 1)[1].decode()
    return None


def run_precheck(unit: str, sha: str, label: str) -> None:
    try:
        gcal.precheck(out_path=None, resume=False, proposal_file=PROPOSAL, proposal_sha256=sha,
                      stub=False, unit=unit)
        print(f"  precheck [{label}]: PASS (no refusal)", flush=True)
    except gcal.GcalRefusal as e:
        print(f"  precheck [{label}]: REFUSED -- {e}", flush=True)
