from math import lgamma, log, exp, ceil
def logsf(N,f,r):
    if r>=1: return 0.0 if N>=f else float('-inf')
    lr,l1=log(r),log(1-r); m=None; terms=[]
    for j in range(f,N+1):
        terms.append(lgamma(N+1)-lgamma(j+1)-lgamma(N-j+1)+j*lr+(N-j)*l1)
    if not terms: return float('-inf')
    m=max(terms); return m+log(sum(exp(t-m) for t in terms))
def n90(f,r):
    lo=f; 
    if exp(logsf(lo,f,r))>=.9: return lo
    hi=f

    while hi-lo>1:
        mid=(lo+hi)//2
        if exp(logsf(mid,f,r))>=.9: hi=mid
        else: lo=mid
    return hi

def pmf(n,p): return [comb(n,k)*p**k*(1-p)**(n-k) for k in range(n+1)]
def crit(n,a):
    # smallest x with P(X>=x)<=a under .5
    tail=0.0
    for x in range(n,-1,-1):
        t=sum(comb(n,k) for k in range(x,n+1))/2**n
        if t>a: return x+1
    return 0
def power(n,ns,pa,ps,alpha,pa2=None,ps2=None):
    pa2=pa if pa2 is None else pa2; ps2=ps if ps2 is None else ps2
    h=n//2; hs=ns//2
    c=crit(n,alpha)
    A1=pmf(h,pa); A2=pmf(h,pa2)
    S1=pmf(hs,ps); S2=pmf(hs,ps2)
    # sham E distribution
    from collections import defaultdict

    for a,pa_ in enumerate(S1):
        for b,pb in enumerate(S2):
            E=abs(Fraction(a+b,ns)-Fraction(1,2))
