## 1. Reading (4): §3.3's sentence on DPO (note :110, row S5#2048)

```swift
import Foundation
import PDFKit
import AppKit
// usage: pdfrender <pdf> <outdir> <scale> <page>...   (1-based pages; writes p<page>_x<scale>.png)
let a = CommandLine.arguments
guard a.count >= 5, let doc = PDFDocument(url: URL(fileURLWithPath: a[1])), let scale = Double(a[3]) else {
    FileHandle.standardError.write("usage: pdfrender <pdf> <outdir> <scale> <page>...\n".data(using: .utf8)!); exit(2)
}
for ps in a[4...] {
    guard let n = Int(ps), n >= 1, n <= doc.pageCount, let page = doc.page(at: n - 1) else { exit(3) }
    let box = page.bounds(for: .mediaBox)
    let w = Int(box.width * scale), h = Int(box.height * scale)
    let cs = CGColorSpaceCreateDeviceRGB()
    guard let ctx = CGContext(data: nil, width: w, height: h, bitsPerComponent: 8, bytesPerRow: 0, space: cs,
                              bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { exit(4) }
    ctx.setFillColor(CGColor(red: 1, green: 1, blue: 1, alpha: 1)); ctx.fill(CGRect(x: 0, y: 0, width: w, height: h))
    ctx.scaleBy(x: scale, y: scale)
    page.draw(with: .mediaBox, to: ctx)
    guard let img = ctx.makeImage() else { exit(5) }
    let rep = NSBitmapImageRep(cgImage: img)
    guard let png = rep.representation(using: .png, properties: [:]) else { exit(6) }
    try! png.write(to: URL(fileURLWithPath: "\(a[2])/p\(n)_x\(a[3]).png"))
}
```
