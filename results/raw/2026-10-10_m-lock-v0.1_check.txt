def package_at(commit) -> str:
    """resume.package_sha256_of's rule over valence_core/m/*.py (top level) at a commit's blobs."""
    out = git("ls-tree", "-z", commit, "valence_core/m/")
    entries = []
    for ent in out.split("\0"):
        if not ent:
            continue
        meta, path = ent.split("\t", 1)
        if meta.split()[1] == "blob" and path.endswith(".py") and path.count("/") == 2:
            entries.append((path, blob(commit, path)))
    h = hashlib.sha256()
    for rel, data in sorted(entries):
        h.update(f"{rel}\n{len(data)}\n".encode("utf-8"))
        h.update(data)
    return h.hexdigest()
