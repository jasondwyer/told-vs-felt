#!/usr/bin/env python3
"""n step 4 -- the per-token diagnostic of the BACKEND extension's cached-versus-fresh discordance.

The seat's answers of 2026-11-18, second sitting, s1.3 item 3: a script in Code's scratch, committed as a raw with its
output, "that, for the 24 reads' sessions and both labels, prints per label-token index the distribution over the 24
of |cached - fresh| per token (min, median, max in nats), and the same for the first token alone against the rest
summed -- no margin, no log-probability, nothing per session". Never of record; no package change; it writes nothing.

WHERE. Once, on n's box (pod emw1e4embiwe9b, the H100 SXM of the BACKEND entry d5d08fa), by the pod session, after the
run of record has ended: in its own scratch copy of the box's checkout at the driver's commit, with this file placed as
scripts/n_bext_diag.py. 5(f)'s quarantine admits the primitive's import from a scripts/n_*.py driver. The committed
driver (scripts/n_sessions.py, imported as a module) supplies the sessions (R1's draw), the ids and the labels of
record ("bare", R2), the weights of record, the pod, cuBLAS, driver-bytes and bridge checks (R6) and the HF backend, so
the 24 reads are the extension's, made by the calls primitive.read makes at an unsteered re-computed read, in its order:
the prefill, S's and U's continuations, S's and U's full forwards, the stage's capture band on each.

WHAT IT PRINTS, and nothing else (cached_t and fresh_t are label token t's log-probability on the two paths, as
primitive.read computes them; every figure is over the reads, in nats):
  1. per label (S, U) and label-token index t: n, and the min, median and max of |cached_t - fresh_t|;
  2. per label: the same for the first token alone, and for the rest summed, |sum over t >= 1 of (cached_t - fresh_t)|;
  3. two counts (Code's addition): the reads whose labels share their first token and whose two full forwards give it
     the same log-probability (on the cached path it is one prefill row for both labels); and the reads whose rel,
     computed here as schema.fidelity_numbers computes it, equals the run of record's fid line to six decimals (the
     run's stdout, by path);
  4. the two paths' timing (Code's addition, for the seat's s1.4 estimate of the fresh path's cost for G-TIME-n): the
     cached read (one prefill, two continuations) and the fresh pair (two full forwards), median and max in ms.
Code's reading of "nothing per session": a cell with fewer than three reads prints its count only.

  python scripts/n_bext_diag.py --run-stdout PATH    the box run: one load, the stage's environment
  python scripts/n_bext_diag.py --selftest           model-free, over the numpy double
"""

import math
import os
import re
import statistics
import subprocess
import sys
import time
from pathlib import Path

_SCRIPTS = Path(__file__).resolve().parent
if _SCRIPTS.name != "scripts" or not (_SCRIPTS.parent / "valence_core" / "n").is_dir():
    raise SystemExit("n_bext_diag.py runs only as <repository>/scripts/n_bext_diag.py (5(f)'s quarantine)")
if str(_SCRIPTS) not in sys.path:
    sys.path.insert(0, str(_SCRIPTS))
import n_sessions as drv  # noqa: E402  (the committed driver; it puts the repository on sys.path)

NAME = "scripts/n_bext_diag.py"
MIN_N = 3
READ_LINE = re.compile(r"^read (\d+) s=(\d+) N1-V none 0 fid=(\d+\.\d{6}) (<=|>) TAU ms=\d+$")


def parse_run_stdout(text: str, plan) -> list:
    """The run of record's read lines: each launch's fid string (six decimals), in launch order. Refused unless the
    stdout holds exactly one read line per launch index of the plan, each of the stage's form, each naming its planned
    session's BK run."""
    got: dict = {}
    for line in text.splitlines():
        m = READ_LINE.match(line)
        if m is None:
            if line.startswith("read "):
                raise SystemExit("the run's stdout holds a read line of another form")
            continue
        i = int(m.group(1))
        if i in got:
            raise SystemExit(f"the run's stdout holds launch index {i} twice")
        got[i] = (int(m.group(2)), m.group(3))
    if sorted(got) != list(range(len(plan))):
        raise SystemExit(f"the run's stdout holds {len(got)} read lines, not one for each of the {len(plan)} launches")
    for i, s in enumerate(plan):
        if got[i][0] != int(s.bk_run):
            raise SystemExit(f"read {i}: the run's line names another session than the plan's")
    return [got[i][1] for i in range(len(plan))]


def measure(backend, plan, ids_of, lab, capture_layers, window: int, clock=time.perf_counter) -> list:
    """Per planned session, in launch order: primitive.read's calls at an unsteered re-computed read, and the per-token
    log-probabilities on both paths. Nothing here is printed."""
    from valence_core.n import primitive, schema
    from valence_core.n.sessions import ids_sha
    out = []
    for k, s in enumerate(plan):
        ids = [int(x) for x in ids_of(s, None)]
        if ids_sha(ids) != s.ids_sha:
            raise SystemExit(f"read {k}: the committed ids are not BK's")
        ls, lu = [int(x) for x in lab(s.cue_S)], [int(x) for x in lab(s.cue_U)]
        if len(ids) + max(len(ls), len(lu)) - 1 > int(window):
            raise SystemExit(f"read {k}: a forward exceeds the stage's window")
        t0 = clock()
        pre = backend.prefill(ids, [], capture_layers)
        cont_s = backend.continue_label(pre, ls)
        cont_u = backend.continue_label(pre, lu)
        t1 = clock()
        rows_fs, _caps_s = backend.full(ids, ls, [], capture_layers)
        rows_fu, _caps_u = backend.full(ids, lu, [], capture_layers)
        t2 = clock()
        c_s, c_u = primitive.score_cached(pre, cont_s, ls), primitive.score_cached(pre, cont_u, lu)
        f_s = [primitive._logp(rows_fs[i], ls[i]) for i in range(len(ls))]
        f_u = [primitive._logp(rows_fu[i], lu[i]) for i in range(len(lu))]
        if not all(math.isfinite(x) for x in c_s + c_u + f_s + f_u):
            raise SystemExit(f"read {k}: a non-finite log-probability")
        lps, lpu, fs, fu = math.fsum(c_s), math.fsum(c_u), math.fsum(f_s), math.fsum(f_u)
        rel = schema.fidelity_numbers(lps - lpu, lps, lpu, fs - fu)["rel"]
        out.append({"S": (c_s, f_s), "U": (c_u, f_u), "rel": rel, "shared_first": ls[0] == lu[0],
                    "first_fresh_equal": ls[0] == lu[0] and f_s[0] == f_u[0],
                    "ms_cached": (t1 - t0) * 1000.0, "ms_fresh": (t2 - t1) * 1000.0})
    return out


def _cell(name: str, xs) -> str:
    if len(xs) < MIN_N:
        return f"{name}: n {len(xs)} (fewer than {MIN_N} reads: the count only)"
    return (f"{name}: n {len(xs)}, |cached - fresh| min {min(xs):.6g}, median {statistics.median(xs):.6g}, "
            f"max {max(xs):.6g} (nats)")


def report_lines(reads: list, run_fids: list) -> list:
    """The diagnostic's output: per label and token index, then the first token alone and the rest summed, then the
    two counts and the timing. No margin, no log-probability, nothing per session."""
    lines = []
    for z in ("S", "U"):
        width = max(len(r[z][0]) for r in reads)
        for t in range(width):
            lines.append(_cell(f"label {z}, token {t}",
                               [abs(r[z][0][t] - r[z][1][t]) for r in reads if len(r[z][0]) > t]))
        lines.append(_cell(f"label {z}, the first token alone", [abs(r[z][0][0] - r[z][1][0]) for r in reads]))
        lines.append(_cell(f"label {z}, the rest summed",
                           [abs(math.fsum(c - f for c, f in zip(r[z][0][1:], r[z][1][1:])))
                            for r in reads if len(r[z][0]) > 1]))
    shared = sum(1 for r in reads if r["shared_first"])
    same = sum(1 for r in reads if r["first_fresh_equal"])
    lines.append(f"first token: shared by both labels in {shared} of {len(reads)} reads; its fresh log-probability "
                 f"identical in the two full forwards in {same} of {shared}")
    agree = sum(1 for r, f in zip(reads, run_fids) if f"{r['rel']:.6f}" == f)
    lines.append(f"identity: the rel computed here equals the run of record's fid line, to six decimals, in {agree} "
                 f"of {len(reads)} reads")
    mc = [r["ms_cached"] for r in reads]
    mf = [r["ms_fresh"] for r in reads]
    lines.append(f"timing over the {len(reads)} reads (ms): the cached read (one prefill, two continuations) median "
                 f"{statistics.median(mc):.1f}, max {max(mc):.1f}; the fresh pair (two full forwards) median "
                 f"{statistics.median(mf):.1f}, max {max(mf):.1f}")
    return lines


def run(env, run_stdout: Path) -> int:
    """The diagnostic, in order: the refusals and every check that needs no model; the load and the bridge again; the
    24 reads; the lines. ``env`` is the driver's ``BextEnv`` (``production_env`` on the box)."""
    from valence_core.m import resume as m_resume
    from valence_core.m.keys import stream_known_answer
    from valence_core.n import ids, schema
    say = env.printer
    pod = env.pod()
    if pod != drv.CHARACTERIZED_POD:
        raise SystemExit(f"this pod is not the characterized one ({drv.CHARACTERIZED_POD}, the BACKEND entry d5d08fa)")
    if os.environ.get("CUBLAS_WORKSPACE_CONFIG") != drv.CHARACTERIZED_PROVENANCE["cublas_workspace_config"]:
        raise SystemExit("CUBLAS_WORKSPACE_CONFIG is not the characterized setting (rule 2)")
    ka = stream_known_answer()
    if ka:
        raise SystemExit(ka[0])
    mine, head = env.driver()
    if head is None or mine != head:
        raise SystemExit("the driver in this copy is not HEAD's committed bytes: the copy is not the commit")
    w = env.weights()
    ss = env.load_sessions()
    loaded = env.load_ids()
    if loaded["manifest_sha256"] != ids.MANIFEST_OF_RECORD:
        raise SystemExit("data/n/ids/MANIFEST.json is not the render of record's")
    plan = drv.bext_sessions(ss)
    fids = parse_run_stdout(Path(run_stdout).read_text(encoding="utf-8"), plan)
    n_layers, d_model, layer = env.shape
    pre = env.env_provenance(w["path"])
    if drv.bridge_identifier(pre) != drv.CHARACTERIZED_IDENTIFIER:
        raise SystemExit("before the load: the environment does not bridge to the characterization's identifier (R6)")
    bid = m_resume.backend_identifier(pre)
    ids_of = ids.ids_of(loaded)
    lab = ids.label_ids(loaded, drv.BEXT_CONVENTION)
    say(f"DIAG, never of record: the BACKEND extension's {len(plan)} reads (labels {drv.BEXT_CONVENTION!r}, d = 0), "
        f"per label token, cached against fresh; pod {pod}")
    t0 = env.clock()
    adapter = env.make_adapter(w["path"])
    load_ms = max(0, int(round((env.clock() - t0) * 1000)))
    prov = adapter.provenance()
    if m_resume.backend_identifier(prov) != bid or drv.bridge_identifier(prov) != drv.CHARACTERIZED_IDENTIFIER:
        raise SystemExit("the loaded adapter's provenance is not the one checked before the load -- refusing")
    if tuple(env.adapter_shape(adapter)) != (n_layers, d_model):
        raise SystemExit("the checkpoint's shape is not the stage's")
    say(f"backend: n's identifier {bid} (window {schema.MAX_WINDOW}); the characterization's "
        f"{drv.CHARACTERIZED_IDENTIFIER} under the default window (R6); load {load_ms} ms")
    reads = measure(env.make_backend(adapter), plan, ids_of, lab, list(range(int(layer) + 1, int(n_layers))),
                    schema.MAX_WINDOW, clock=env.clock)
    for line in report_lines(reads, fids):
        say(line)
    return 0


def _tracked() -> bool:
    got = subprocess.run(["git", "-C", str(drv.REPO), "ls-files", "--error-unmatch", NAME], capture_output=True,
                         check=False)
    return got.returncode == 0


def selftest() -> int:
    """Model-free, over the numpy double: the arithmetic against primitive.read's own, the report's lines, the run's
    stdout parser, and the flow behind the driver's seams, refusals first."""
    from valence_core.n import ids, primitive, schema, testing
    results: list = []

    def chk(name, ok, detail=""):
        results.append(bool(ok))
        if not ok:
            print(f"FAIL {name}: {str(detail)[:300]}")

    def refused(fn) -> bool:
        try:
            fn()
        except SystemExit as e:
            return e.code not in (0, None)
        return False

    LS, LU = [5, 9, 17], [5, 11, 2]
    flow = [testing.toy_session(r) for r in range(drv.BEXT_N)]
    fss = [s for s, _i in flow]
    by = {s.session_id: list(i) for s, i in flow}
    labels = {fss[0].cue_S: LS, fss[0].cue_U: LU}

    def ids_of(s, _told):
        return by[s.session_id]

    def lab(cue):
        return labels[cue]
    plan = drv.bext_sessions(fss)
    caps = [3, 4, 5]

    r0 = measure(testing.CausalDouble(), plan, ids_of, lab, caps, schema.MAX_WINDOW)
    chk("diag: 24 reads, in the plan's order", len(r0) == drv.BEXT_N and plan == fss, len(r0))
    chk("diag: without a fresh bias every per-token difference and every rel is 0 (the double's paths agree)",
        all(c == f for r in r0 for z in "SU" for c, f in zip(*r[z])) and all(r["rel"] == 0.0 for r in r0))
    chk("diag: ...the first token is shared, its fresh value identical in the two full forwards, in 24 of 24",
        sum(r["shared_first"] for r in r0) == 24 and sum(r["first_fresh_equal"] for r in r0) == 24)

    be1 = testing.CausalDouble(fresh_bias=(9, 0.3))
    rb = measure(be1, plan, ids_of, lab, caps, schema.MAX_WINDOW)
    be2 = testing.CausalDouble(fresh_bias=(9, 0.3))
    spec = primitive.ReadSpec(schema.SHADOW_ARM, schema.SHADOW_DIRECTION, schema.SHADOW_DOSE, 0.0, None)
    same = True
    for i, (s, r) in enumerate(zip(plan, rb)):
        rec, _c = primitive.read(be2, s, by[s.session_id], s.ids_sha, spec, LS, LU, caps,
                                 fidelity_reason="dose-rule", shadow=None, identity_bound=None, launch_index=i,
                                 capture_row=i)
        f = rec["fidelity"]
        same = same and (rec["logprob_S_tokens"] == r["S"][0] and rec["logprob_U_tokens"] == r["U"][0]
                         and f["fresh_logprob_S"] == math.fsum(r["S"][1])
                         and f["fresh_logprob_U"] == math.fsum(r["U"][1]) and f["rel"] == r["rel"])
    chk("diag: with a fresh bias, each read's cached tokens, fresh sums and rel are primitive.read's exactly", same)
    chk("diag: ...its backend calls are primitive.read's, call for call (prefill, two continuations, two fulls, "
        "the same lengths)", be1.calls == be2.calls
        and [c[0] for c in be1.calls[:5]] == ["prefill", "continue", "continue", "full", "full"], be1.calls[:5])
    chk("diag: ...and the bias shows in some per-token difference and some rel",
        any(c != f for r in rb for z in "SU" for c, f in zip(*r[z])) and any(r["rel"] > 0 for r in rb))
    chk("diag: ...the bias on a continuation token's id leaves the first token's fresh value equal in both forwards",
        sum(r["first_fresh_equal"] for r in rb) == 24)

    fids = [f"{r['rel']:.6f}" for r in rb]
    lines = report_lines(rb, fids)
    chk("diag: the report: per label 3 token lines, the first token and the rest, then the two counts and the timing",
        len(lines) == 2 * (3 + 2) + 3 and lines[0].startswith("label S, token 0: n 24, |cached - fresh| min ")
        and lines[3].startswith("label S, the first token alone: n 24")
        and lines[4].startswith("label S, the rest summed: n 24") and lines[5].startswith("label U, token 0: n 24")
        and lines[10].startswith("first token: shared by both labels in 24 of 24 reads; its fresh log-probability "
                                 "identical in the two full forwards in 24 of 24")
        and lines[12].startswith("timing over the 24 reads (ms): the cached read"), lines)
    rest_s = [abs((r["S"][0][1] - r["S"][1][1]) + (r["S"][0][2] - r["S"][1][2])) for r in rb]
    chk("diag: ...its first token alone is token 0's cell, and the rest is |(c1 - f1) + (c2 - f2)| over the reads",
        lines[3].split(": n ", 1)[1] == lines[0].split(": n ", 1)[1]
        and lines[4] == _cell("label S, the rest summed", rest_s) and max(rest_s) > 0, (lines[3], lines[4]))
    chk("diag: ...the identity count is 24 of 24 against the run's own lines",
        lines[11] == "identity: the rel computed here equals the run of record's fid line, to six decimals, in 24 of "
                     "24 reads", lines[11])
    bad = list(fids)
    bad[3] = "9.999999"
    chk("diag: ...and 23 of 24 when one line differs", "in 23 of 24 reads" in report_lines(rb, bad)[11])
    few = report_lines(rb[:2], fids[:2])
    chk("diag: a cell of fewer than 3 reads prints its count only",
        all(x.endswith("(fewer than 3 reads: the count only)") for x in few[:10]), few[:2])
    chk("diag: no line carries a log-probability or a margin: the cached and fresh values themselves never print",
        not any(f"{v:.6g}" in x for x in lines for r in rb for z in "SU" for v in r[z][0] + r[z][1] if abs(v) > 1e-3),
        "")

    def stdout_for(fs, sess=None, extra=()):
        sess = plan if sess is None else sess
        reads = [f"read {i} s={s.bk_run} N1-V none 0 fid={f} {'>' if float(f) > schema.TAU else '<='} TAU ms=7"
                 for i, (s, f) in enumerate(zip(sess, fs))]
        return "\n".join(["WEIGHTS: toy", "G-VER (toy):"] + reads + list(extra)
                         + ["PAUSE fidelity-binomial at the run's end"]) + "\n"
    chk("diag: the run's stdout gives its fid strings in launch order", parse_run_stdout(stdout_for(fids), plan) == fids)
    two = [testing.toy_session(0)[0], testing.toy_session(5)[0]]
    smoke = ("read 0 s=0 N1-V none 0 fid=0.014660 <= TAU ms=973\nread 1 s=5 N1-V none 0 fid=0.086070 > TAU ms=712\n"
             "PAUSE fidelity-binomial at the run's end\n")
    chk("diag: ...the smoke's two read lines, as the box printed them (0eb7671), parse to their fid strings",
        parse_run_stdout(smoke, two) == ["0.014660", "0.086070"])
    swapped = list(plan)
    swapped[0], swapped[1] = swapped[1], swapped[0]
    for name, text in (("23 read lines", stdout_for(fids[:23])),
                       ("a launch index twice", stdout_for(fids, extra=[f"read 0 s=0 N1-V none 0 fid={fids[0]} <= TAU "
                                                                       "ms=7"])),
                       ("a line naming another session", stdout_for(fids, sess=swapped)),
                       ("a read line of another form", stdout_for(fids, extra=["read 24 s=9 N1-V none 0 "
                                                                              "fid=nonfinite ms=7"]))):
        chk(f"diag: the run's stdout is refused: {name}", refused(lambda t=text: parse_run_stdout(t, plan)))

    saved = os.environ.get("CUBLAS_WORKSPACE_CONFIG")
    tmp_out = Path(os.environ.get("TMPDIR", "/tmp")) / f"n_bext_diag_selftest_{os.getpid()}.txt"
    try:
        os.environ["CUBLAS_WORKSPACE_CONFIG"] = drv.CHARACTERIZED_PROVENANCE["cublas_workspace_config"]
        tmp_out.write_text(stdout_for(fids), encoding="utf-8")
        loaded_t = {"manifest_sha256": ids.MANIFEST_OF_RECORD,
                    "by_session": {s.session_id: {"ids": list(i)} for s, i in flow}, "by_variant": {},
                    "candidates": {"bare": {"name": "bare", "kept": True, "labels": dict(labels)}}}
        prov_t = dict(drv.CHARACTERIZED_PROVENANCE, path="(toy)", choice_window_declared=schema.MAX_WINDOW)
        made: list = []

        class StubAdapter:
            def __init__(self, prov):
                self._prov = prov

            def provenance(self):
                return dict(self._prov)

        def env_for(**over):
            lines_: list = []

            def mk(p):
                made.append(p)
                return StubAdapter(prov_t)
            base = dict(weights=lambda: {"repo": "toy", "revision": "0" * 12, "sha256": "5" * 64, "n_files": 1,
                                         "path": "(toy)"},
                        load_sessions=lambda: list(fss), load_ids=lambda: loaded_t,
                        gver=lambda s_, l_, p_: {"status": "PASS"}, env_provenance=lambda p: dict(prov_t),
                        make_adapter=mk, adapter_shape=lambda a: (6, 12),
                        make_backend=lambda a: testing.CausalDouble(fresh_bias=(9, 0.3)), bk_table=lambda f_: {},
                        pod=lambda: drv.CHARACTERIZED_POD, driver=lambda: (b"drv", b"drv"), shape=(6, 12, 2),
                        printer=lines_.append)
            loaded_prov = over.pop("loaded_prov", None)
            base.update(over)
            if loaded_prov is not None:
                base["make_adapter"] = lambda p: (made.append(p), StubAdapter(loaded_prov))[1]
            return drv.BextEnv(**base), lines_

        env1, out1 = env_for()
        rc = run(env1, tmp_out)
        chk("diag: the flow over the double: exit 0, one load, the DIAG and backend lines, then the report",
            rc == 0 and len(made) == 1 and out1[0].startswith("DIAG, never of record: the BACKEND extension's 24 reads")
            and out1[1].startswith("backend: n's identifier ") and out1[2:-1] == lines[:-1]
            and out1[-1].startswith("timing over the 24 reads (ms): the cached read"), out1[:3])
        other = dict(prov_t, gpu="another GPU")
        for name, kw in (("another pod", dict(pod=lambda: "another")), ("no pod", dict(pod=lambda: None)),
                         ("a driver that is not HEAD's", dict(driver=lambda: (b"a", b"b"))),
                         ("no committed driver", dict(driver=lambda: (b"a", None))),
                         ("ids that are not the render of record's",
                          dict(load_ids=lambda: dict(loaded_t, manifest_sha256="0" * 64))),
                         ("an environment that does not bridge", dict(env_provenance=lambda p: dict(other)))):
            n0 = len(made)
            envx, _o = env_for(**kw)
            chk(f"diag: refused before the load: {name}", refused(lambda e=envx: run(e, tmp_out)) and len(made) == n0)
        n0 = len(made)
        envb, _o = env_for()
        tmp_bad = tmp_out.with_name(tmp_out.name + ".bad")
        tmp_bad.write_text(stdout_for(fids[:23]), encoding="utf-8")
        chk("diag: refused before the load: a run stdout without its 24 read lines",
            refused(lambda: run(envb, tmp_bad)) and len(made) == n0)
        tmp_bad.unlink()
        os.environ["CUBLAS_WORKSPACE_CONFIG"] = ":16:8"
        envc, _o = env_for()
        chk("diag: refused before the load: another cuBLAS setting", refused(lambda: run(envc, tmp_out))
            and len(made) == n0)
        os.environ["CUBLAS_WORKSPACE_CONFIG"] = drv.CHARACTERIZED_PROVENANCE["cublas_workspace_config"]
        envd, _o = env_for(loaded_prov=other)
        chk("diag: refused after the load, before any read: a loaded provenance other than the checked one",
            refused(lambda: run(envd, tmp_out)) and len(made) == n0 + 1)
    finally:
        if saved is None:
            os.environ.pop("CUBLAS_WORKSPACE_CONFIG", None)
        else:
            os.environ["CUBLAS_WORKSPACE_CONFIG"] = saved
        if tmp_out.exists():
            tmp_out.unlink()
    chk("diag: the environment is as it was", os.environ.get("CUBLAS_WORKSPACE_CONFIG") == saved)
    n_ok = sum(results)
    print(f"  {n_ok}/{len(results)} n_bext_diag selftest checks pass")
    return 0 if n_ok == len(results) else 1


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(prog="n_bext_diag.py")
    g = ap.add_mutually_exclusive_group(required=True)
    g.add_argument("--selftest", action="store_true")
    g.add_argument("--run-stdout", default=None)
    a = ap.parse_args(argv)
    if a.selftest:
        return selftest()
    if _tracked():
        raise SystemExit(f"{NAME} is tracked in this checkout: the diagnostic is never of record")
    return run(drv.production_env(), Path(a.run_stdout))


if __name__ == "__main__":
    sys.exit(main())
