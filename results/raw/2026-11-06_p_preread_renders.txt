"""PROPOSAL-p v0.1 §1.2 reproduced: corr.corrupt_m's own walk, its eligible set restricted to §1.1's frame words.

Run from the repository root: python -B <this file>. Reads data/f/pools.json and the rosters of record through

FUNCTION_WORDS = {"a", "an", "the", "of", "in", "on", "is", "are", "does", "do", "has", "have", "many", "or",
                  "than", "by", "to", "and", "which", "what", "how"}


def words_of(text):
    """(start, end, word, quoted) for each maximal run of ASCII letters; quoted = inside a single-quoted span."""
    out, i, in_quote = [], 0, False
    while i < len(text):
        ch = text[i]
        if ch == "'":
            in_quote = not in_quote
            i += 1
            continue
        if ("a" <= ch <= "z") or ("A" <= ch <= "Z"):
            j = i
            while j < len(text) and (("a" <= text[j] <= "z") or ("A" <= text[j] <= "Z")):
                j += 1
            out.append((i, j, text[i:j], in_quote))
            i = j
            continue
        i += 1
    return out


def frame_words(text, case_insensitive=True):
    ws = words_of(text)
    fr = []
    for idx, (s, e, w, q) in enumerate(ws):
        if q:
            continue
        key = w.lower() if case_insensitive else w
        if idx == 0 or key in FUNCTION_WORDS:
            fr.append((s, e, w))
    return fr


def eligible_p(text, case_insensitive=True):
    spans = frame_words(text, case_insensitive)
    return [p for p in corr._eligible_m(text) if any(s <= p < e for s, e, _ in spans)]


def corrupt_p(text, i, rosters):
    orig = corr._eligible_m
    corr._eligible_m = lambda t: [p for p in orig(t) if any(s <= p < e for s, e, _ in frame_words(t))]
    try:
        return corr.corrupt_m(text, i, rosters)
    finally:
        corr._eligible_m = orig


def k_at(n, rate_pct):
    return min(max((rate_pct * n + 50) // 100, 3), n // 2)
