def crit(n, p0, alpha):
    """Smallest c with P(Bin(n, p0) >= c) <= alpha, by a descending upper-tail sum; None if none <= n."""
    if (n, p0, alpha) not in _CRIT:
        _CRIT[(n, p0, alpha)] = _crit(n, p0, alpha)
    return _CRIT[(n, p0, alpha)]


def _crit(n, p0, alpha):
    a, b = p0.numerator, p0.denominator
    total = b ** n
    tail, best = 0, None
    for c in range(n, -1, -1):
        tail += comb(n, c) * a ** c * (b - a) ** (n - c)
        if tail * alpha.denominator <= alpha.numerator * total:
            best = c
        else:
            break
    return best


def seq_false_pause(ns, p0, alpha):
    """P(some look j: cumulative count >= crit(n_j)), binomial increments at p0, by a DP over survivors."""
    a, b = p0.numerator, p0.denominator
    surv, prev = {0: 1}, 0                         # weights over b**prev
    for n in ns:
        c = crit(n, p0, alpha)
        m = n - prev
        inc = [comb(m, i) * a ** i * (b - a) ** (m - i) for i in range(m + 1)]
        nxt = {}
        for s, w in surv.items():
            for i, wi in enumerate(inc):
                t = s + i
                if c is not None and t >= c:
                    continue
                nxt[t] = nxt.get(t, 0) + w * wi
        surv, prev = nxt, n
    return 1 - Fraction(sum(surv.values()), b ** prev)


def n90(floor, k, nq):
    """Smallest N with P(Bin(N, k/nq) >= floor) >= 9/10, i.e. 10 * P(X <= floor - 1) <= 1, by bisection."""
    a, b = k, nq - k
    if b == 0:
        return floor

    def ok(n):
        return 10 * sum(comb(n, x) * a ** x * b ** (n - x) for x in range(min(floor, n + 1))) <= nq ** n
    lo, hi = floor, floor
    while not ok(hi):
        lo, hi = hi, hi * 2
    while lo < hi:
        mid = (lo + hi) // 2
        if ok(mid):
            hi = mid
        else:
            lo = mid + 1
    return lo


def launch(floor, unit, k, nq):
    up = (115 * floor + 99) // 100
    nn = n90(floor, k, nq)
    big = max(up, nn)
    return up, nn, ((big + unit - 1) // unit) * unit


def looks_of(stop):
    counts = {a: 0 for a in ARMS}
    total, out = 0, []
    while any(counts[a] < stop[a] for a in ARMS):
        live = [a for a in ARMS if counts[a] < stop[a]]
        nxt = min(live, key=lambda a: (counts[a], ARMS.index(a)))
        counts[nxt] += 1
        total += 1
        if total % EVERY == 0:
            out.append([total] + [counts[a] for a in ARMS])
    if not out or out[-1][0] != total:
        out.append([total] + [counts[a] for a in ARMS])
    return out


def main(argv):
    rec = json.load(open(argv[1], encoding="utf-8"))
    seat_raw = open(argv[2], "rb").read()
    check("schema mf3-gpwr-v1, unit Q-F3, path uncached",
          (rec["schema"], rec["unit"], rec["path"]) == ("mf3-gpwr-v1", "Q-F3", "uncached"))
    inp = rec["inputs"]
    check("inputs: k 29 of 40, r 29/40, the floors, the balance units, looks every 40",
          (inp["k"], inp["n_q"], inp["r"], inp["floors"], inp["balance_units"], inp["look_every"])
          == (29, 40, "29/40", FLOOR, UNIT, 40))
    L = {}
    for a in ARMS:
        up, nn, l_ = launch(FLOOR[a], UNIT[a], K, NQ)
        r_ = rec["launch"]["rule"][a]
        L[a] = l_
        check("%s: ceil(1.15 floor) %d, N90 %d and L %d by an exact binomial search (k 29/40)" % (a, up, nn, l_),
              (r_["floor"], r_["ceil_1.15_floor"], r_["N90"], r_["L"], rec["launch"]["L"][a]) == (FLOOR[a], up, nn,
                                                                                                    l_, l_))
    total = sum(L.values())
    check("total %d, looks ceil(total/40) = %d" % (total, -(-total // EVERY)),
          (rec["launch"]["total"], rec["launch"]["looks"]) == (total, -(-total // EVERY)) and total == 832)
    bad, same = [], []
    memo = {}
    for k in range(1, NQ + 1):
        row = {}
        for a in ARMS:
            key = (FLOOR[a], UNIT[a], k)
            if key not in memo:
                memo[key] = launch(FLOOR[a], UNIT[a], k, NQ)[2]
            row[a] = memo[key]
        t = sum(row.values())
        got = rec["launch"]["rows_by_k"][str(k)]
        if {**row, "total": t, "looks": -(-t // EVERY)} != got:
            bad.append(k)
        if row == L:
            same.append(k)
    check("rows k = 1..40: L per arm, total and looks, each recomputed", not bad and len(rec["launch"]["rows_by_k"]) == 40,
          bad)
    check("the same row at k = %s" % same, rec["launch"]["k_with_this_row"] == same)
    stops = {
        "projection": dict(L),
        "sham_at_floor_r": {**L, **{s: -(-FLOOR[s] * NQ // K) for s in SHAM}},
        "sham_at_floor_min": {**L, **{s: FLOOR[s] for s in SHAM}},
        "sham_realised": dict(L),
    }
    for name, stop in stops.items():
        cs = rec["cases"][name]
        lk = looks_of(stop)
        got = [[x["launches"]] + [x[a] for a in ARMS] for x in cs["looks"]]
        flags = [x["look"] for x in cs["looks"]] == list(range(1, len(lk) + 1)) and \
            [bool(x.get("end")) for x in cs["looks"]] == [False] * (len(lk) - 1) + [True]
        check("%s: stop %s; %d looks by the campaign rule" % (name, "/".join(str(stop[a]) for a in ARMS), len(lk)),
              cs["stop"] == stop and got == lk and flags and cs["n_looks"] == len(lk))
        for cls, (p0, dom, per, kind) in BARS.items():
            ex = [Fraction(sum(row[1 + ARMS.index(a)] for a in dom)) * (REALISED[kind] if name == "sham_realised"
                                                                          else per) for row in lk]
            ns = [half_up(x) for x in ex]
            if name != "sham_realised" and any(x.denominator != 1 for x in ex):
                check("%s %s: whole units" % (name, cls), False)
            cj = [crit(n, p0, ALPHA) for n in ns]
            fp = seq_false_pause(ns, p0, ALPHA)
            bf = cs["bars"][cls]
            ok = (bf["n_j"], bf["crit_j"], Fraction(bf["false_pause"]["exact"]), bf["false_pause"]["display"]) == \
                (ns, cj, fp, dec(fp, 4))
            if name == "sham_realised":
                ok = ok and bf["n_j_exact"] == ["%d/%d" % (x.numerator, x.denominator) for x in ex]
            check("%s %s: n_j, crit_j and the sequential false pause %s (p0 %s)" % (name, cls, dec(fp, 4), p0), ok)
        if name == "sham_realised":
            check("sham_realised: no guard (the guards count launches)", "guards" not in cs and "guard_unit" not in cs)
            continue
        for c, s in PAIRS:
            g = cs["guards"][c]
            n1n2 = [[row[1 + ARMS.index(c)], row[1 + ARMS.index(s)]] for row in lk]
            n1 = [x[0] for x in n1n2]
            ac = [crit(n, ABS_P0, ALPHA) for n in n1]
            afp = seq_false_pause(n1, ABS_P0, ALPHA)
            at = g["at_common_rate_p0"]
            check("%s guard %s/%s: looks, abs crit_j, the absolute half %s at p0 1/5" % (name, c, s, dec(afp, 4)),
                  (g["looks_n1_n2"], g["abs_crit_j"], Fraction(at["abs"]["exact"]), at["abs"]["display"], at["p"])
                  == (n1n2, ac, afp, dec(afp, 4), "1/5"))
            un = Fraction(at["union"]["exact"])
            check("%s guard %s/%s: the pausing figure is the union, at least each half" % (name, c, s),
                  g["guard_figure"]["figure"] == "union" and g["guard_figure"]["at_common_rate_p0"] == at["union"]
                  and un >= max(afp, Fraction(at["fisher"]["exact"])) and at["union"]["display"] == dec(un, 4))
        surv = Fraction(1)
        for c, _ in PAIRS:
            surv *= 1 - Fraction(cs["guards"][c]["at_common_rate_p0"]["union"]["exact"])
        gu = cs["guard_unit"]
        check("%s: the unit's guard figure, 1 - prod(1 - union) over the three pairs, %s" % (name, dec(1 - surv, 4)),
              Fraction(gu["exact"]) == 1 - surv and gu["display"] == dec(1 - surv, 4)
              and gu["pairs"] == [list(p) for p in PAIRS] and gu["figure_per_pair"] == "union")
    pc, jj = rec["cases"]["projection"], rec["joint"]
    check("the joint: each bar's own on the projection, the guards' unions and the unit's figure beside",
          jj["figures"] == {cls: pc["bars"][cls]["false_pause"] for cls in BARS}
          and jj["guards_beside"] == {c: pc["guards"][c]["at_common_rate_p0"]["union"] for c, _ in PAIRS}
          and jj["guard_unit_beside"] == pc["guard_unit"])
    hb = rec["hours"]
    hrs = [Fraction(total) * b / 3600 for b in BASIS]
    check("hours at m:230's basis: %d x 641.6-713.6 s = %s-%s h (%d-%d; m:238's 148-165)"
          % (total, dec(hrs[0], 2), dec(hrs[1], 2), half_up(hrs[0]), half_up(hrs[1])),
          [Fraction(h["exact"]) for h in hb["hours"]] == hrs
          and [(h["display"], h["whole"]) for h in hb["hours"]] == [(dec(h, 2), str(half_up(h))) for h in hrs]
          and (hb["basis_line"], hb["launches"], hb["m238_column"]) == ("m:230", 832, "148–165")
          and "%d–%d" % (half_up(hrs[0]), half_up(hrs[1])) == hb["m238_column"])
    pw = rec["power"]
    check("power at the floors: A1 and A2 displayed >= .85 and gated, A3 reported (not gated); no n raised",
          all(Fraction(pw["arms"][a]["exact"]) >= Fraction(17, 20) and pw["arms"][a]["gated"] for a in ("A1", "A2"))
          and not pw["arms"]["A3"]["gated"] and pw["n_raised"] is False
          and all(pw["arms"][a]["display"] == dec(Fraction(pw["arms"][a]["exact"]), 3) for a in ("A1", "A2", "A3"))
          and [(pw["arms"][a]["n"], pw["arms"][a]["mS"]) for a in ("A1", "A2", "A3")] == [(80, 32), (144, 56),
                                                                                          (144, 56)])
    bz = rec["beside_A5"]
    check("§A.5's rows beside, each verbatim in the seat's raw (sha256 %s...)" % SEAT_SHA[:8],
          hashlib.sha256(seat_raw).hexdigest() == SEAT_SHA and bz["sha256"] == SEAT_SHA and len(bz["rows"]) == 6
          and all(r.encode() in seat_raw for r in bz["rows"]) and bz["launches"] == 832)
    check("the cap: none set (the PI's)", "not set by this program" in rec["cap"] and "cap" not in rec["hours"])
    if len(argv) > 3:
        pv = rec["provenance"]
        check("provenance: HEAD and the script's sha256 are the run's",
              (pv["head"], pv["script_sha256"], pv["isolated"]) == (argv[3], argv[4], True))
    print("verify: %s (%d failed)" % ("PASS" if not FAILS else "FAIL", len(FAILS)))
    return 0 if not FAILS else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv))
