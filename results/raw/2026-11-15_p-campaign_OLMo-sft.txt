# The PROPOSAL-p campaign under each checkpoint's LOCK-p v1.0 (038bf3e), on the seat's answers of 2026-11-14

# Chain rule: continue on exit 0 with no STOP line, or on PAUSE (exit 3, its PAUSE line, no STOP line), each with its

if [ ! -e "$OUT/stamps.txt" ]; then
  echo "launch $(date -u +%Y-%m-%dT%H:%M:%SZ) HEAD=$(git rev-parse HEAD) RUNPOD_POD_ID=$RUNPOD_POD_ID" >> "$OUT/stamps.txt"

  echo "launch (a later invocation of the batch: no stamp) HEAD=$(git rev-parse HEAD) RUNPOD_POD_ID=$RUNPOD_POD_ID" >> "$OUT/stamps.txt"

# ==== wait_campaign.sh, the laptop's end-wait (sha256 cfdc111940f9b03d4866b6e84103796fe5745d01e9329a87a01e27fae8549c90) ====
#!/bin/bash
# The end-wait for the campaign batch on the wrapper's PID alone (the list of record v2, item 5; O2-2's fix: the PID
# paired with its command so a reused PID cannot pass): no read of the stage's stamps, stdout, checkpoints or reports.
# Prints nothing but its own end line. Usage: wait_campaign.sh <wrapper pid>; BOX_ADDR and BOX_PORT from the environment.
# Its command text names none of the words the laptop's process checks look for. wait_qual.sh's form.
set -u
PID=$1
: "${BOX_ADDR:?}" "${BOX_PORT:?}"
fails=0
while true; do
  ssh -q -i "$HOME/.ssh/id_runpod" -p "$BOX_PORT" -o BatchMode=yes -o ConnectTimeout=20 "root@$BOX_ADDR" \
    "ps -p $PID -o args= 2>/dev/null | grep -q '[r]un_campaign_p[.]sh'"
  rc=$?
  if [ "$rc" -eq 1 ]; then echo "the wrapper (PID $PID) has ended"; exit 0; fi
  if [ "$rc" -ne 0 ]; then
    fails=$((fails+1))
    [ "$fails" -ge 30 ] && { echo "the box did not answer 30 times in a row; the wait stops, the run is not known to have ended"; exit 2; }
    sleep 60
    continue
  fi
  fails=0
  sleep 120
done
