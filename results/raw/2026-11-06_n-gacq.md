### `gacq_verify.py`

```python
#!/usr/bin/env python3
"""G-ACQ (n) -- verify the verbatim download against the hub's own listing, and build the manifest entry.

Usage (on the box, model-free, CPU only):
    python -I gacq_verify.py <hub listing api_rev.json> <staging dir> <out json>

Checks, each reported:
  (1) the expected file set: every hub file matching the acquisition's allow patterns
      (*.safetensors, *.json, *.txt, *.model -- scripts/box_acquire_olmo.py's) is present, and nothing else
      is (the downloader's own .cache/ metadata folder is listed and set aside: it is removed before the move);
  (2) per file: size equals the hub's; an LFS file's sha256 equals the hub's LFS sha256; a non-LFS file's
      git blob sha1 equals the hub's blobId;
  (3) the shards' safetensors headers: every tensor's dtype, the parameter count, and the tensor-name set
      against model.safetensors.index.json's weight_map (each tensor in exactly the shard the index names);
  (4) per-file sha256 and bytes, and the aggregate exactly as scripts/box_hash_checkpoints.py defines it:
      sha256 over "".join(f"{name}:{sha256}" for name in sorted(files)).
Writes the proposed data/olmo/manifest.json entry (key olmo2_instruct, path /workspace/models/olmo2_instruct)
to <out json>. Exits 0 only if (1)-(3) all pass.
"""
from __future__ import annotations

import fnmatch
import hashlib
import json
import os
import struct
import sys
import time

ALLOW = ["*.safetensors", "*.json", "*.txt", "*.model"]
CHUNK = 1 << 22
KEY = "olmo2_instruct"
FINAL_PATH = "/workspace/models/olmo2_instruct"
REPO = "allenai/OLMo-2-0325-32B-Instruct"
REVISION_FULL = "b96024342a77a69aa0dda815c3454a671f477463"
BYTES_PER = {"BF16": 2, "F16": 2, "F32": 4, "F64": 8, "I64": 8, "I32": 4, "I16": 2, "I8": 1, "U8": 1, "BOOL": 1}


def hashes(p: str) -> tuple[str, str, int]:
    """(sha256, git blob sha1, bytes) in one pass."""
    n = os.path.getsize(p)
    s256 = hashlib.sha256()
    g1 = hashlib.sha1(b"blob %d\0" % n)
    with open(p, "rb") as f:
        while (b := f.read(CHUNK)):
            s256.update(b)
            g1.update(b)
    return s256.hexdigest(), g1.hexdigest(), n


def st_header(p: str) -> dict:
    with open(p, "rb") as f:
        (n,) = struct.unpack("<Q", f.read(8))
        return json.loads(f.read(n))


def main() -> int:
    listing_p, stage, out_p = sys.argv[1:4]
    listing = json.load(open(listing_p))
    if listing.get("sha") != REVISION_FULL:
        print(f"ABORT: listing is for {listing.get('sha')}, not {REVISION_FULL}")
        return 2
    hub = {s["rfilename"]: s for s in listing["siblings"]
           if any(fnmatch.fnmatch(s["rfilename"], a) for a in ALLOW)}
    problems: list[str] = []

    # (1) the file set
    on_disk = sorted(os.listdir(stage))
    set_aside = [x for x in on_disk if x == ".cache"]
    present = [x for x in on_disk if x != ".cache"]
    for x in present:
        if os.path.isdir(os.path.join(stage, x)):
            problems.append(f"(1) unexpected directory {x}")
    missing = sorted(set(hub) - set(present))
    extra = sorted(set(present) - set(hub))
    problems += [f"(1) MISSING {x}" for x in missing] + [f"(1) EXTRA {x}" for x in extra]
    print(f"(1) hub files under the allow patterns: {len(hub)}; present: {len(present)}; "
          f"missing {len(missing)}; extra {len(extra)}; set aside: {set_aside or 'none'}")

    # (2) + (4) per file
    files: dict[str, dict] = {}
    t0 = time.time()
    for fn in sorted(set(hub) & set(present)):
        s = hub[fn]
        sha, blob, n = hashes(os.path.join(stage, fn))
        files[fn] = {"sha256": sha, "bytes": n}
        if n != s.get("size"):
            problems.append(f"(2) {fn}: size {n} != hub {s.get('size')}")
        if s.get("lfs"):
            ok = sha == s["lfs"]["sha256"]
            what = f"sha256 vs hub LFS sha256 {'==' if ok else '!='}"
        else:
            ok = blob == s.get("blobId")
            what = f"git blob sha1 vs hub blobId {'==' if ok else '!='}"
        if not ok:
            problems.append(f"(2) {fn}: {what}")
        print(f"  {fn:<36} {n:>14,}  sha256 {sha[:16]}…  {what}", flush=True)
    print(f"(2) hashed {len(files)} files in {time.time() - t0:.0f}s")

    # (3) safetensors headers
    shards = sorted(f for f in files if f.endswith(".safetensors"))
    dtypes: dict[str, int] = {}
    n_params = 0
    name_to_shard: dict[str, str] = {}
    for fn in shards:
        h = st_header(os.path.join(stage, fn))
        for k, v in h.items():
            if k == "__metadata__":
                continue
            numel = 1
            for d in v["shape"]:
                numel *= d
            n_params += numel
            dtypes[v["dtype"]] = dtypes.get(v["dtype"], 0) + numel
            if k in name_to_shard:
                problems.append(f"(3) tensor {k} in two shards")
            name_to_shard[k] = fn
            lo, hi = v["data_offsets"]
            if hi - lo != numel * BYTES_PER.get(v["dtype"], 0):
                problems.append(f"(3) {fn}:{k}: byte span {hi - lo} != numel x dtype size")
    idx = json.load(open(os.path.join(stage, "model.safetensors.index.json")))["weight_map"]
    if set(idx) != set(name_to_shard):
        problems.append(f"(3) index tensor set != shard headers' ({len(idx)} vs {len(name_to_shard)})")
    wrong = [k for k, v in idx.items() if name_to_shard.get(k) != v]
    if wrong:
        problems.append(f"(3) {len(wrong)} tensors in a shard other than the index's (first: {wrong[0]})")
    if set(dtypes) != {"BF16"}:
        problems.append(f"(3) dtypes {dtypes}, not pure BF16")
    hub_params = (listing.get("safetensors") or {}).get("total")
    if n_params != hub_params:
        problems.append(f"(3) n_params {n_params} != the hub's safetensors total {hub_params}")
    print(f"(3) shards {len(shards)}; tensors {len(name_to_shard)}; dtypes {dtypes}; n_params {n_params:,} "
          f"(hub {hub_params:,}); index weight_map {len(idx)} entries, {len(wrong)} misplaced")

    agg = hashlib.sha256("".join(f"{k}:{files[k]['sha256']}" for k in sorted(files)).encode()).hexdigest()
    size_gb = round(sum(v["bytes"] for v in files.values()) / 1e9, 1)
    done = None
    try:
        done = open(os.path.join(os.path.dirname(stage.rstrip("/")), "gacq_download.exit")).read().split()[1]
    except (OSError, IndexError):
        pass
    entry = {"repo": REPO, "revision": REVISION_FULL[:12], "dtype": "bfloat16", "n_params": n_params,
             "size_gb": size_gb, "path": FINAL_PATH, "acquired_utc": done,
             "files": files, "files_aggregate_sha256": agg, "n_files": len(files)}
    json.dump({"key": KEY, "entry": entry, "problems": problems, "set_aside": set_aside,
               "revision_full": REVISION_FULL}, open(out_p, "w"), indent=1)
    print(f"(4) {len(files)} files, {size_gb} GB, aggregate {agg}")
    print()
    if problems:
        print(f"G-ACQ VERIFY FAIL -- {len(problems)} problem(s):")
        for x in problems:
            print("  " + x)
        return 1
    print("G-ACQ VERIFY PASS -- every file matches the hub's listing; pure BF16; index and headers agree")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

### `gacq_hooksite.py` (attempt 2's script. Attempt 1's differed in three places: its constructor line was `ad = OLMoHFAdapter(PATH)  # deterministic=True: refuses without CUBLAS_WORKSPACE_CONFIG (env.sh sets it)`; it lacked the four comment lines and the `cap = …` line before `t0 = time.time()`; and it lacked the record keys `max_position_embeddings`, `choice_window_passed` and `runpod_pod_id`)

```python
#!/usr/bin/env python3
"""G-ACQ (n) -- the OLMo-2 hook-site identity check (PROPOSAL-n v0.2 s3 G-ACQ: "the OLMo-2 hook site (the
residual after decoder layer 32) verified against olmo_hf.py's capture positions by a one-forward identity
check").

One forward of /workspace/models/olmo2_instruct through valence_core.adapters.olmo_hf.OLMoHFAdapter
(deterministic, bf16, the adapter's own load path), with the adapter's own capture hooks (_make_hooks, no
injection) on every decoder layer AND output_hidden_states=True on that same call. For each layer index l
the adapter's capture (the output of model.model.layers[l]) is compared with transformers' hidden_states
entry for it: hidden_states[l + 1] for l < n_layers - 1 (hidden_states[0] is the embedding output); the last
layer's entry is reported separately, since transformers may apply the final norm to the last element.

PASS iff every l < n_layers - 1 is bitwise equal (torch.equal). Descriptive, deciding nothing: the residual
L2 norms at indices 31 and 32 (mean over positions), for G-VEC's R check (25.49 within 2%), which is where
BK's "layer 32 of 64" convention is settled; and the ids and the text, so the check is reproducible.

Usage (on the box, after `. /workspace/env.sh`): python gacq_hooksite.py <out json>
"""
from __future__ import annotations

import hashlib
import json
import os
import sys
import time

sys.path.insert(0, os.environ.get("JSPACE", "/workspace/repo/jspace-lab"))

import torch  # noqa: E402

from valence_core.adapters.olmo_hf import OLMoHFAdapter  # noqa: E402

PATH = "/workspace/models/olmo2_instruct"
# A fixed sentence drawn from no pool, prompt or corpus of this program or BK's (the check is about positions).
TEXT = "A single forward pass over one fixed sentence is enough to check where a hook reads."


def main() -> int:
    out_p = sys.argv[1]
    # The adapter's DEFAULT_CHOICE_WINDOW (8192) is OLMo-3's; OLMo-2-32B has 4096 positions, and the adapter refuses a
    # declared window above the checkpoint's cap (attempt 1 of this check, 2026-11-07T03:17Z, refused there, before
    # any forward). This check reads no choice window: it passes the checkpoint's own cap, from its config.json.
    # n's own windows are declared by the n package (PROPOSAL-n s8 step 6; rule 28(d)'s standing form (c)).
    cap = int(json.load(open(os.path.join(PATH, "config.json")))["max_position_embeddings"])
    t0 = time.time()
    ad = OLMoHFAdapter(PATH, choice_window=cap)  # deterministic=True: refuses without CUBLAS_WORKSPACE_CONFIG
    t_load = time.time() - t0
    L = ad.n_layers
    ids = ad.encode(TEXT)
    store: dict[int, torch.Tensor] = {}
    handles = ad._make_hooks([], range(L), store, [0])
    try:
        x = torch.tensor([ids], device=ad.device)
        with torch.inference_mode():
            out = ad._model(input_ids=x, use_cache=False, output_hidden_states=True)
    finally:
        for h in handles:
            h.remove()
    hs = out.hidden_states
    rows, fails = [], []
    for li in range(L):
        a = store[li][0]
        b = hs[li + 1][0] if li + 1 < len(hs) else None
        if b is None:
            rows.append({"layer": li, "equal": None, "max_abs": None})
            continue
        eq = bool(torch.equal(a, b))
        mad = float((a.float() - b.float()).abs().max().item())
        rows.append({"layer": li, "equal": eq, "max_abs": mad})
        if li < L - 1 and not eq:
            fails.append(li)
    norms = {str(li): float(store[li][0].float().norm(dim=-1).mean().item()) for li in (31, 32) if li in store}
    res = {
        "path": PATH, "model_class": type(ad._model).__name__,
        "transformers": __import__("transformers").__version__, "torch": torch.__version__,
        "n_layers": L, "hidden_size": int(ad._model.config.hidden_size),
        "max_position_embeddings": cap, "choice_window_passed": cap,
        "runpod_pod_id": os.environ.get("RUNPOD_POD_ID"),
        "len_hidden_states": len(hs), "text": TEXT, "ids": ids,
        "ids_sha16": hashlib.sha256(json.dumps(ids).encode()).hexdigest()[:16],
        "rows": rows, "fails": fails, "last_layer_equal": rows[-1]["equal"],
        "mean_resid_norm_at": norms, "load_s": round(t_load, 1),
        "verdict": "PASS" if not fails and len(hs) == L + 1 else "FAIL",
    }
    json.dump(res, open(out_p, "w"), indent=1)
    print(f"model {res['model_class']} | n_layers {L} | hidden {res['hidden_size']} | len(hidden_states) {len(hs)}")
    print(f"ids ({len(ids)}) sha16 {res['ids_sha16']} | load {t_load:.0f}s")
    for r in rows:
        if r["layer"] in (0, 30, 31, 32, 33, L - 2, L - 1) or r["equal"] is False:
            print(f"  layer {r['layer']:>2}: capture == hidden_states[{r['layer'] + 1}] -> {r['equal']} "
                  f"(max |diff| {r['max_abs']})")
    print(f"mean residual norm (descriptive): index 31 {norms.get('31'):.4f}; index 32 {norms.get('32'):.4f}")
    print(f"HOOK-SITE {res['verdict']} -- {L - 1 - len(fails)}/{L - 1} layers bitwise equal below the last; "
          f"last layer equal: {res['last_layer_equal']}")
    return 0 if res["verdict"] == "PASS" else 1


if __name__ == "__main__":
    sys.exit(main())
```
