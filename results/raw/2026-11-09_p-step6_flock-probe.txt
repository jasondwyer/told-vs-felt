# flock had run there before. This probe, run on the pod before any lock is cut, asks: does the lock exclude another
# process and a second file descriptor in the same process, and is it released on close? It wrote only under

# local_lock=none: the client sends its locks to the server (none is local to the pod). Exclusion across two pods on

# Reading: on the pod's nfs4 volume the lock excludes another process and a second descriptor in the same process

"""Model-free flock probe for the box's /workspace volume (round 4's m-2): does an exclusive, non-blocking flock
exclude a second open file description (another process, and a second fd in the same process), and is it released
on close? Writes only under /workspace/flock_probe/ (outside the repository)."""
import fcntl
import os
import subprocess
import sys

D = "/workspace/flock_probe"
os.makedirs(D, exist_ok=True)
P = os.path.join(D, "probe.lock")
CHILD = ("import fcntl, os, sys\n"
         "fd = os.open(sys.argv[1], os.O_RDWR | os.O_CREAT, 0o644)\n"
         "try:\n"
         "    fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)\n"
         "    print('acquired')\n"
         "except BlockingIOError:\n"
         "    print('BlockingIOError')\n"
         "except OSError as e:\n"
         "    print('OSError', e.errno, os.strerror(e.errno))\n")


def child():
    r = subprocess.run([sys.executable, "-c", CHILD, P], capture_output=True, text=True)
    return (r.stdout.strip() or r.stderr.strip()[-200:])


out = {}
fd = os.open(P, os.O_RDWR | os.O_CREAT, 0o644)
try:
    fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
    out["parent_acquire"] = "acquired"
except OSError as e:
    out["parent_acquire"] = f"OSError {e.errno} {os.strerror(e.errno)}"
out["child_while_held"] = child()
fd2 = os.open(P, os.O_RDWR)
try:
    fcntl.flock(fd2, fcntl.LOCK_EX | fcntl.LOCK_NB)
    out["same_process_second_fd_while_held"] = "acquired"
except BlockingIOError:
    out["same_process_second_fd_while_held"] = "BlockingIOError"
except OSError as e:
    out["same_process_second_fd_while_held"] = f"OSError {e.errno} {os.strerror(e.errno)}"
finally:
    os.close(fd2)
os.close(fd)
out["child_after_release"] = child()
st = os.statvfs(D)
out["statvfs_f_fsid"] = getattr(st, "f_fsid", None)
mounts = [ln for ln in open("/proc/mounts") if " /workspace " in ln]
out["mount"] = [" ".join(ln.split()[:3]) for ln in mounts]
for k, v in out.items():
    print(f"{k}: {v}")
