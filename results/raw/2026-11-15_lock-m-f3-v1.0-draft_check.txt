# --- PIN texts (v02_gen.py's extraction rule)
def pin_at(lines, i):
    if lines[i].lstrip().startswith("#"):
        j = i
        while (j + 1 < len(lines) and lines[j + 1].lstrip().startswith("#")
               and not lines[j + 1].lstrip().startswith("# PIN")):
            j += 1
        return " ".join(l.lstrip()[1:].strip() for l in lines[i:j + 1]), "comment"
    j = i
    while j + 1 < len(lines) and lines[j + 1].strip() and '"""' not in lines[j]:
        j += 1
    return " ".join(l.strip() for l in lines[i:j + 1]).replace('"""', "").strip(), "docstring"


def table_of(lines, marker):
    out = []
    for i, l in enumerate(lines):
        if marker in l:
            txt, kind = pin_at(lines, i)
            out.append((i + 1, kind, txt[txt.index("PIN (Code"):]))
    return out
