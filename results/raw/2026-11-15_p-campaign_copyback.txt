#!/bin/bash
# Copy back the campaign batch's outputs (rule 27): the files the stage left untracked under data/p on the box, by tar over
# ssh into the laptop repo, each sha256-compared. Refuses anything outside the campaign's names in data/p/ckpt and
# data/p/reports, and any local file already present. Prints names and digests only; opens no file's contents (the
# checkpoints are pick-bearing, the call logs withheld). The wrapper's own outputs go to the scratch directory $1, for
# the raw. Usage: copyback_campaign.sh <scratch dir>; the pod's address and port come from the environment (BOX_ADDR,
# BOX_PORT), never from a committed file.
set -u
SCR=$1
REPO=<repo>
: "${BOX_ADDR:?}" "${BOX_PORT:?}"
SSH=(ssh -i "$HOME/.ssh/id_runpod" -p "$BOX_PORT" -o BatchMode=yes -o ConnectTimeout=20 "root@$BOX_ADDR")
LIST=$("${SSH[@]}" 'cd /workspace/repo/jspace-lab && git status --porcelain --untracked-files=all | sed -n "s/^?? //p"') \
  || { echo "COPYBACK: listing failed"; exit 10; }
[ -n "$LIST" ] || { echo "COPYBACK: the box has no untracked output"; exit 11; }
echo "$LIST" | grep -vE '^data/p/(ckpt|reports)/OLMo-(sft|dpo|instruct)_campaign(\.json|_p-calls\.jsonl|_p-STOPPED\.json|_adjudication\.json|_report\.json|_reads\.json|_p-report\.json)$' \
  && { echo "COPYBACK: unexpected untracked files above"; exit 12; }
for f in $LIST; do [ -e "$REPO/$f" ] && { echo "COPYBACK: $f already exists locally"; exit 13; }; done
"${SSH[@]}" "cd /workspace/repo/jspace-lab && tar cf - $(echo $LIST)" | tar xmf - -C "$REPO" || { echo "COPYBACK: tar failed"; exit 14; }
BOX=$("${SSH[@]}" "cd /workspace/repo/jspace-lab && sha256sum $(echo $LIST)") || { echo "COPYBACK: box hashes failed"; exit 15; }
bad=0
while read -r h f; do
  l=$(shasum -a 256 "$REPO/$f" | awk '{print $1}')
  if [ "$h" = "$l" ]; then echo "identical $h $f"; else echo "DIFFERS box $h laptop $l $f"; bad=1; fi
done <<< "$BOX"
mkdir -p "$SCR"
"${SSH[@]}" 'cd /workspace/campaign_p && tar cf - stamps.txt d_lines.txt wrapper-*.out campaign_p-*.txt precheck_campaign.txt precheck_campaign.exit' | tar xmf - -C "$SCR" || { echo "COPYBACK: wrapper outputs failed"; exit 16; }
"${SSH[@]}" 'cd /workspace/campaign_p && sha256sum stamps.txt d_lines.txt wrapper-*.out campaign_p-*.txt precheck_campaign.txt precheck_campaign.exit' > "$SCR/box_sha256.txt"
exit $bad
