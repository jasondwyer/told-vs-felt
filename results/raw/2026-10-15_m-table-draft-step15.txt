"""Segment the preprint of record into sentences: (line, ordinal, section, text). Deterministic.
A line is split after [.!?] (plus any closing *, _, ), ], ", ', ’, ”, superscript digits) when followed by
whitespace and then a capital letter, digit, *, [, (, “, ", _, Δ, π or §, unless the period ends a declared
abbreviation or an initial. Headings, table rows and list items keep their markers; a bold run-in head
ending in '.**' is its own sentence."""
import json, re, sys
ABBR = ("et al.", "e.g.", "i.e.", "cf.", "vs.", "Fig.", "No.", "approx.", "Dr.", "St.", "pp.", "Vol.", "ed.", "eds.",
        "Jr.", "Ph.D.", "U.S.", "n.", "al.", "resp.")
END = re.compile(r"[.!?][*_)\]\"'’”¹²³⁴⁵⁶⁷⁸⁹⁰]*\s+(?=[A-Z0-9*\[(“\"_Δπ§])")
def split(line):
    out, start = [], 0
    for m in END.finditer(line):
        seg = line[start:m.end()].rstrip()
        head = line[start:m.start() + 1]
        if any(re.search(r"(?:^|[\s(\[])" + re.escape(a) + "$", head) for a in ABBR):
            continue
        if re.search(r"(?:^|[\s(])[A-Z]\.$", head):          # an initial: "J." "M."
            continue
        if re.fullmatch(r"\d+\.", head.strip()):            # a list item's marker stays with its text
            continue
        out.append(seg)
        start = m.end()
    tail = line[start:].rstrip()
    if tail:
        out.append(tail)
    return out
text = open(sys.argv[1], encoding="utf-8").read()
lines = text.split("\n")
section = "front"
sub = None
rows = []
for i, l in enumerate(lines, 1):
    if not l.strip():
        continue
    if l.startswith("#"):
        section = l.lstrip("#").strip()
        sub = None
        rows.append({"line": i, "sentence": 1, "section": section, "text": l})
        continue
    m = re.match(r"\*\*(8\.\d) ", l)
    sub = m.group(1) if m else sub              # a bold run-in sub-section label holds until the next heading
    sec = section + (" / " + sub if sub else "")
    for k, s in enumerate(split(l), 1):
        rows.append({"line": i, "sentence": k, "section": sec, "text": s})
json.dump(rows, open(sys.argv[2], "w", encoding="utf-8"), ensure_ascii=False, indent=0)
print(len(rows), "sentences on", len({r['line'] for r in rows}), "lines")
